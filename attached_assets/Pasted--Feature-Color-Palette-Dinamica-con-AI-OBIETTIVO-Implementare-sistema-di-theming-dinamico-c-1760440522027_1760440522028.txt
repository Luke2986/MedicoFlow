# 🎨 Feature: Color Palette Dinamica con AI

## OBIETTIVO
Implementare sistema di theming dinamico che permette agli ambulatori di personalizzare MedicoFlow con i colori del proprio brand, estraendo automaticamente la palette da un URL tramite AI.

**Features:**
- Sezione "Aspetto" nelle impostazioni
- Color palette default (attuale blu medicale)
- Estrazione AI automatica colori da URL sito
- Anteprima live before/after
- Applicazione dinamica via CSS variables
- Persistenza preferenze per ambulatorio

## 📄 FILE 2: lib/ai/color-extractor.ts

```typescript
import Anthropic from '@anthropic-ai/sdk'

const anthropic = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY
})

export interface ColorPalette {
  primary: string
  primaryLight: string
  primaryDark: string
  accent: string
  success: string
  warning: string
  error: string
}

export async function extractColorsFromUrl(url: string): Promise<ColorPalette> {
  try {
    // 1. Fetch HTML del sito (con timeout)
    const controller = new AbortController()
    const timeout = setTimeout(() => controller.abort(), 10000) // 10s timeout
    
    const response = await fetch(url, {
      signal: controller.signal,
      headers: {
        'User-Agent': 'MedicoFlow Color Extractor/1.0'
      }
    })
    clearTimeout(timeout)

    if (!response.ok) {
      throw new Error(`Failed to fetch URL: ${response.status}`)
    }

    const html = await response.text()

    // 2. Estrai sezione rilevante (primi 50KB per performance)
    const htmlSnippet = html.substring(0, 50000)

    // 3. Usa Claude per analizzare ed estrarre colori
    const message = await anthropic.messages.create({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 1024,
      messages: [
        {
          role: 'user',
          content: `Analizza questo HTML e CSS inline ed estrai la color palette principale del brand.

HTML SNIPPET:
\`\`\`html
${htmlSnippet}
\`\`\`

Identifica:
1. PRIMARY COLOR: Il colore principale del brand (logo, CTA, header)
2. ACCENT COLOR: Colore secondario/accento
3. SUCCESS/WARNING/ERROR: Colori per stati (se presenti, altrimenti suggerisci appropriati)

Rispondi SOLO con JSON in questo formato:
{
  "primary": "#RRGGBB",
  "primaryLight": "#RRGGBB",
  "primaryDark": "#RRGGBB", 
  "accent": "#RRGGBB",
  "success": "#RRGGBB",
  "warning": "#RRGGBB",
  "error": "#RRGGBB"
}

IMPORTANTE: 
- Tutti i colori devono essere in formato esadecimale #RRGGBB
- primaryLight deve essere 20% più chiaro di primary
- primaryDark deve essere 20% più scuro di primary
- Se non trovi colori espliciti, estrapola dal contesto del brand
- Assicurati che i colori abbiano buon contrasto per accessibilità`
        }
      ]
    })

    const content = message.content[0]
    if (content.type !== 'text') {
      throw new Error('Unexpected response type from Claude')
    }

    // 4. Parse risposta JSON
    const jsonText = content.text.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim()
    const colors: ColorPalette = JSON.parse(jsonText)

    // 5. Valida formato colori
    const hexPattern = /^#[0-9A-Fa-f]{6}$/
    const requiredKeys: (keyof ColorPalette)[] = [
      'primary', 'primaryLight', 'primaryDark', 'accent', 
      'success', 'warning', 'error'
    ]

    for (const key of requiredKeys) {
      if (!colors[key] || !hexPattern.test(colors[key])) {
        throw new Error(`Invalid color format for ${key}: ${colors[key]}`)
      }
    }

    return colors
  } catch (error) {
    console.error('Error extracting colors:', error)
    throw new Error('Impossibile estrarre i colori dal sito. Verifica che l\'URL sia corretto e accessibile.')
  }
}

export function getDefaultPalette(): ColorPalette {
  return {
    primary: '#2563eb',      // blue-600
    primaryLight: '#3b82f6', // blue-500
    primaryDark: '#1e40af',  // blue-700
    accent: '#10b981',       // green-500
    success: '#22c55e',      // green-500
    warning: '#f59e0b',      // amber-500
    error: '#ef4444'         // red-500
  }
}
```

## 📄 FILE 3: app/api/theme/extract/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { extractColorsFromUrl } from '@/lib/ai/color-extractor'
import { getCurrentUser } from '@/lib/auth'

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const body = await request.json()
    const { url } = body

    if (!url) {
      return NextResponse.json({ error: 'URL mancante' }, { status: 400 })
    }

    // Valida URL
    try {
      new URL(url)
    } catch {
      return NextResponse.json({ error: 'URL non valido' }, { status: 400 })
    }

    // Estrai colori con AI
    const palette = await extractColorsFromUrl(url)

    return NextResponse.json({ 
      success: true,
      palette,
      sourceUrl: url
    })
  } catch (error: any) {
    console.error('Error extracting colors:', error)
    return NextResponse.json({ 
      error: error.message || 'Errore durante l\'estrazione dei colori' 
    }, { status: 500 })
  }
}
```

## 📄 FILE 4: app/api/theme/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { getDefaultPalette } from '@/lib/ai/color-extractor'

export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const settings = await prisma.themeSettings.findUnique({
      where: { ambulatorioId: user.ambulatorioId }
    })

    // Se non esiste, return default
    if (!settings) {
      const defaultPalette = getDefaultPalette()
      return NextResponse.json({
        ...defaultPalette,
        sourceType: 'default',
        sourceUrl: null
      })
    }

    return NextResponse.json({
      primary: settings.primaryColor,
      primaryLight: settings.primaryLight,
      primaryDark: settings.primaryDark,
      accent: settings.accentColor,
      success: settings.successColor,
      warning: settings.warningColor,
      error: settings.errorColor,
      sourceType: settings.sourceType,
      sourceUrl: settings.sourceUrl
    })
  } catch (error) {
    console.error('Error fetching theme:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function PUT(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const body = await request.json()
    const { 
      primary, primaryLight, primaryDark, 
      accent, success, warning, error: errorColor,
      sourceType, sourceUrl 
    } = body

    const settings = await prisma.themeSettings.upsert({
      where: { ambulatorioId: user.ambulatorioId },
      create: {
        ambulatorioId: user.ambulatorioId,
        primaryColor: primary,
        primaryLight: primaryLight,
        primaryDark: primaryDark,
        accentColor: accent,
        successColor: success,
        warningColor: warning,
        errorColor: errorColor,
        sourceType: sourceType || 'custom',
        sourceUrl: sourceUrl
      },
      update: {
        primaryColor: primary,
        primaryLight: primaryLight,
        primaryDark: primaryDark,
        accentColor: accent,
        successColor: success,
        warningColor: warning,
        errorColor: errorColor,
        sourceType: sourceType || 'custom',
        sourceUrl: sourceUrl
      }
    })

    return NextResponse.json({ success: true, settings })
  } catch (error) {
    console.error('Error saving theme:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function DELETE(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    // Reset a default eliminando il record
    await prisma.themeSettings.deleteMany({
      where: { ambulatorioId: user.ambulatorioId }
    })

    return NextResponse.json({ success: true })
  } catch (error) {
    console.error('Error resetting theme:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

## 📄 FILE 5: components/theme/ThemeProvider.tsx

```typescript
'use client'

import { createContext, useContext, useEffect, useState } from 'react'

interface ThemeContextType {
  palette: {
    primary: string
    primaryLight: string
    primaryDark: string
    accent: string
    success: string
    warning: string
    error: string
  }
  reloadTheme: () => Promise<void>
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined)

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [palette, setPalette] = useState({
    primary: '#2563eb',
    primaryLight: '#3b82f6',
    primaryDark: '#1e40af',
    accent: '#10b981',
    success: '#22c55e',
    warning: '#f59e0b',
    error: '#ef4444'
  })

  const loadTheme = async () => {
    try {
      const res = await fetch('/api/theme')
      if (res.ok) {
        const data = await res.json()
        setPalette(data)
        applyTheme(data)
      }
    } catch (error) {
      console.error('Error loading theme:', error)
    }
  }

  const applyTheme = (colors: typeof palette) => {
    const root = document.documentElement
    
    // Applica CSS variables
    root.style.setProperty('--color-primary', colors.primary)
    root.style.setProperty('--color-primary-light', colors.primaryLight)
    root.style.setProperty('--color-primary-dark', colors.primaryDark)
    root.style.setProperty('--color-accent', colors.accent)
    root.style.setProperty('--color-success', colors.success)
    root.style.setProperty('--color-warning', colors.warning)
    root.style.setProperty('--color-error', colors.error)
  }

  useEffect(() => {
    loadTheme()
  }, [])

  return (
    <ThemeContext.Provider value={{ palette, reloadTheme: loadTheme }}>
      {children}
    </ThemeContext.Provider>
  )
}

export function useTheme() {
  const context = useContext(ThemeContext)
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return context
}
```

## 📄 FILE 6: app/dashboard/impostazioni/aspetto/page.tsx

```typescript
'use client'

import { useState } from 'react'
import { ArrowLeft, Palette, Loader2, Eye, Download, RotateCcw, Sparkles } from 'lucide-react'
import Link from 'next/link'
import { useTheme } from '@/components/theme/ThemeProvider'

export default function AspettoPage() {
  const { palette, reloadTheme } = useTheme()
  const [loading, setLoading] = useState(false)
  const [extracting, setExtracting] = useState(false)
  const [previewPalette, setPreviewPalette] = useState(palette)
  const [siteUrl, setSiteUrl] = useState('')
  const [error, setError] = useState('')
  const [success, setSuccess] = useState(false)
  const [sourceType, setSourceType] = useState<'default' | 'url' | 'custom'>('default')

  const handleExtractColors = async () => {
    if (!siteUrl.trim()) {
      setError('Inserisci un URL valido')
      return
    }

    setExtracting(true)
    setError('')

    try {
      const res = await fetch('/api/theme/extract', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ url: siteUrl })
      })

      const data = await res.json()

      if (!res.ok) {
        throw new Error(data.error || 'Errore durante estrazione')
      }

      setPreviewPalette(data.palette)
      setSourceType('url')
      setSuccess(true)
      setTimeout(() => setSuccess(false), 3000)
    } catch (err: any) {
      setError(err.message)
    } finally {
      setExtracting(false)
    }
  }

  const handleSaveTheme = async () => {
    setLoading(true)
    setError('')

    try {
      const res = await fetch('/api/theme', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          primary: previewPalette.primary,
          primaryLight: previewPalette.primaryLight,
          primaryDark: previewPalette.primaryDark,
          accent: previewPalette.accent,
          success: previewPalette.success,
          warning: previewPalette.warning,
          error: previewPalette.error,
          sourceType,
          sourceUrl: sourceType === 'url' ? siteUrl : null
        })
      })

      if (!res.ok) {
        throw new Error('Errore durante il salvataggio')
      }

      await reloadTheme()
      setSuccess(true)
      setTimeout(() => setSuccess(false), 3000)
    } catch (err: any) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  const handleResetToDefault = async () => {
    setLoading(true)
    setError('')

    try {
      const res = await fetch('/api/theme', {
        method: 'DELETE'
      })

      if (!res.ok) {
        throw new Error('Errore durante il reset')
      }

      await reloadTheme()
      setPreviewPalette({
        primary: '#2563eb',
        primaryLight: '#3b82f6',
        primaryDark: '#1e40af',
        accent: '#10b981',
        success: '#22c55e',
        warning: '#f59e0b',
        error: '#ef4444'
      })
      setSourceType('default')
      setSiteUrl('')
      setSuccess(true)
      setTimeout(() => setSuccess(false), 3000)
    } catch (err: any) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="space-y-6">
      <div className="flex items-center gap-4">
        <Link
          href="/dashboard/impostazioni"
          className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
        >
          <ArrowLeft className="w-5 h-5" />
        </Link>
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Aspetto</h1>
          <p className="text-gray-600 mt-1">Personalizza i colori del tuo ambulatorio</p>
        </div>
      </div>

      {error && (
        <div className="bg-red-50 border border-red-200 text-red-700 px-4 py-3 rounded-lg">
          {error}
        </div>
      )}

      {success && (
        <div className="bg-green-50 border border-green-200 text-green-700 px-4 py-3 rounded-lg">
          ✅ Modifiche salvate con successo
        </div>
      )}

      {/* Metodo 1: Colori Default */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
        <div className="flex items-center justify-between mb-4">
          <div>
            <h2 className="text-xl font-semibold text-gray-900 flex items-center gap-2">
              <Palette className="w-5 h-5" />
              Palette Default
            </h2>
            <p className="text-sm text-gray-600 mt-1">
              Usa i colori medicali professionali predefiniti
            </p>
          </div>
          <button
            onClick={handleResetToDefault}
            disabled={loading}
            className="flex items-center gap-2 px-4 py-2 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 disabled:opacity-50 transition-colors"
          >
            <RotateCcw className="w-4 h-4" />
            Reset Default
          </button>
        </div>
      </div>

      {/* Metodo 2: Estrai da URL */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
        <div className="mb-4">
          <h2 className="text-xl font-semibold text-gray-900 flex items-center gap-2">
            <Sparkles className="w-5 h-5 text-purple-600" />
            Estrai Colori dal Tuo Sito
            <span className="px-2 py-0.5 bg-purple-100 text-purple-700 text-xs font-semibold rounded">
              AI
            </span>
          </h2>
          <p className="text-sm text-gray-600 mt-1">
            Inserisci l'URL del tuo sito e l'AI estrarrà automaticamente i colori del brand
          </p>
        </div>

        <div className="flex gap-3">
          <input
            type="url"
            value={siteUrl}
            onChange={(e) => setSiteUrl(e.target.value)}
            placeholder="https://tuosito.it"
            className="flex-1 px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
          />
          <button
            onClick={handleExtractColors}
            disabled={extracting || !siteUrl.trim()}
            className="flex items-center gap-2 px-6 py-2 bg-purple-600 text-white rounded-lg hover:bg-purple-700 disabled:opacity-50 transition-colors"
          >
            {extracting ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                Estrazione...
              </>
            ) : (
              <>
                <Download className="w-4 h-4" />
                Estrai Colori
              </>
            )}
          </button>
        </div>

        {extracting && (
          <div className="mt-4 p-4 bg-purple-50 border border-purple-200 rounded-lg">
            <div className="flex items-center gap-3">
              <Loader2 className="w-5 h-5 text-purple-600 animate-spin" />
              <div>
                <p className="font-medium text-purple-900">Analisi in corso...</p>
                <p className="text-sm text-purple-700">
                  L'AI sta analizzando il sito e identificando i colori del brand
                </p>
              </div>
            </div>
          </div>
        )}
      </div>

      {/* Anteprima Palette */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
        <div className="flex items-center gap-2 mb-4">
          <Eye className="w-5 h-5 text-gray-600" />
          <h2 className="text-xl font-semibold text-gray-900">Anteprima Palette</h2>
        </div>

        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
          {Object.entries(previewPalette).map(([name, color]) => (
            <div key={name} className="space-y-2">
              <div
                className="w-full h-20 rounded-lg border-2 border-gray-200 shadow-sm"
                style={{ backgroundColor: color }}
              />
              <div>
                <p className="text-sm font-medium text-gray-900 capitalize">
                  {name.replace(/([A-Z])/g, ' $1').trim()}
                </p>
                <p className="text-xs text-gray-500 font-mono">{color}</p>
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* Demo Preview */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
        <h2 className="text-xl font-semibold text-gray-900 mb-4">
          Anteprima Live
        </h2>
        <div className="space-y-3">
          <button
            style={{ backgroundColor: previewPalette.primary }}
            className="px-6 py-3 text-white rounded-lg font-medium shadow-sm"
          >
            Button Primary
          </button>
          <button
            style={{ backgroundColor: previewPalette.accent }}
            className="px-6 py-3 text-white rounded-lg font-medium shadow-sm ml-3"
          >
            Button Accent
          </button>
          <button
            style={{ backgroundColor: previewPalette.success }}
            className="px-6 py-3 text-white rounded-lg font-medium shadow-sm ml-3"
          >
            Success
          </button>
        </div>
      </div>

      {/* Actions */}
      <div className="flex gap-4">
        <button
          onClick={handleSaveTheme}
          disabled={loading}
          className="flex-1 flex items-center justify-center gap-2 px-6 py-3 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50 transition-colors"

            </>
          ) : (
            <>
              <Palette className="w-4 h-4" />

          href="/dashboard/impostazioni"
          className="px-6 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
        >
          Annulla
        </Link>
      </div>

      {/* Info Box */}
      <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
```
