Crea setup iniziale per web app gestionale ambulatorio medico italiano "MedicoFlow".

STACK:
- Next.js 14 con App Router
- TypeScript (strict mode)
- Prisma ORM + PostgreSQL
- Tailwind CSS
- shadcn/ui per componenti

SETUP RICHIESTO:

1. Inizializza progetto Next.js 14 con TypeScript
2. Installa dipendenze base:
   npm install prisma @prisma/client
   npm install -D tailwindcss postcss autoprefixer
   npm install date-fns
   npm install lucide-react

3. Configura Prisma:
   - Inizializza Prisma con PostgreSQL
   - Crea file prisma/schema.prisma con questo schema:
```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id            String   @id @default(cuid())
  email         String   @unique
  name          String
  role          String   @default("medico")
  ambulatorioId String
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
}

model Ambulatorio {
  id              String   @id @default(cuid())
  nome            String
  indirizzo       String
  citta           String
  provincia       String
  telefono        String
  email           String
  piva            String
  regione         String
  specializzazione String
  createdAt       DateTime @default(now())
}

model Paziente {
  id              String   @id @default(cuid())
  nome            String
  cognome         String
  codiceFiscale   String   @unique
  dataNascita     DateTime
  luogoNascita    String
  sesso           String
  indirizzo       String
  citta           String
  provincia       String
  cap             String
  telefono        String
  email           String?
  note            String?
  ambulatorioId   String
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@index([cognome, nome])
}

model Appuntamento {
  id            String   @id @default(cuid())
  pazienteId    String
  medicoId      String
  dataOra       DateTime
  durata        Int      @default(30)
  tipo          String
  stato         String   @default("confermato")
  motivoVisita  String?
  note          String?
  ambulatorioId String
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
  
  @@index([dataOra])
  @@index([pazienteId])
}
