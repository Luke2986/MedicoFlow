export type PlanKey = keyof typeof PLANS

export function hasFeature(
  subscription: { piano: string; features: string[] },
  feature: string
): boolean {
  const plan = PLANS[subscription.piano as PlanKey]
  return plan?.features.includes(feature) || subscription.features.includes(feature)
}

export function canUpgrade(currentPlan: PlanKey): boolean {
  const order: PlanKey[] = ['BASE', 'PROFESSIONAL', 'ENTERPRISE']
  const currentIndex = order.indexOf(currentPlan)
  return currentIndex < order.length - 1
}

export function getNextPlan(currentPlan: PlanKey): PlanKey | null {
  const order: PlanKey[] = ['BASE', 'PROFESSIONAL', 'ENTERPRISE']
  const currentIndex = order.indexOf(currentPlan)
  return currentIndex < order.length - 1 ? order[currentIndex + 1] : null
}

6. Crea helper per ottenere subscription - lib/subscription.ts:

import { prisma } from './prisma'
import { getCurrentUser } from './auth'
import { hasFeature } from './features'

export async function getCurrentSubscription() {
  const user = await getCurrentUser()
  if (!user) return null

  const ambulatorio = await prisma.ambulatorio.findUnique({
    where: { id: user.ambulatorioId },
    include: { subscription: true }
  })

  return ambulatorio?.subscription || null
}

export async function checkFeatureAccess(feature: string): Promise<boolean> {
  const subscription = await getCurrentSubscription()
  if (!subscription) return false
  
  return hasFeature(subscription, feature)
}

9. Crea API route per gestione subscription - app/api/subscription/route.ts:

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { PLANS } from '@/lib/features'

export async function GET() {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const subscription = await prisma.subscription.findUnique({
      where: { ambulatorioId: user.ambulatorioId }
    })

    return NextResponse.json(subscription)
  } catch (error) {
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function PUT(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const { piano } = await request.json()
    
    if (!PLANS[piano as keyof typeof PLANS]) {
      return NextResponse.json({ error: 'Piano non valido' }, { status: 400 })
    }

    const planDetails = PLANS[piano as keyof typeof PLANS]

    const subscription = await prisma.subscription.update({
      where: { ambulatorioId: user.ambulatorioId },
      data: {
        piano,
        prezzoMensile: planDetails.price * 100,
        features: planDetails.features
      }
    })

    return NextResponse.json(subscription)
  } catch (error) {
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

interface AgentSearchResult {
  paziente: any
  relevanceScore: number
  reasoning: string
}

export async function aiAgentSearch(
  query: string,
  pazienti: any[],
  context?: string
): Promise<AgentSearchResult[]> {
  // Usa Claude API di Anthropic (disponibile in Replit)
  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': process.env.ANTHROPIC_API_KEY || '',
      'anthropic-version': '2023-06-01'
    },
    body: JSON.stringify({
      model: 'claude-3-haiku-20240307',
      max_tokens: 2000,
      messages: [{
        role: 'user',
        content: `Sei un assistente intelligente per la ricerca di pazienti in un ambulatorio medico italiano.

QUERY UTENTE: "${query}"

DATABASE PAZIENTI (JSON):
${JSON.stringify(pazienti, null, 2)}

ISTRUZIONI:
1. Analizza la query in linguaggio naturale
2. Trova i pazienti più rilevanti
3. Gestisci typo e variazioni (es. "Maro" = "Mario")
4. Comprendi sinonimi e contesto italiano
5. Ritorna max 5 risultati ordinati per rilevanza

FORMATO RISPOSTA (JSON):
{
  "results": [
    {
      "pazienteId": "id_paziente",
      "relevanceScore": 0.95,
      "reasoning": "Spiega perché è rilevante"
    }
  ]
}

Rispondi SOLO con JSON valido, nessun testo aggiuntivo.`
      }]
    })
  })

  const data = await response.json()
  const content = data.content[0].text
  
  // Parse risposta JSON
  const parsed = JSON.parse(content)
  
  // Map risultati con pazienti completi
  return parsed.results.map((r: any) => ({
    paziente: pazienti.find(p => p.id === r.pazienteId),
    relevanceScore: r.relevanceScore,
    reasoning: r.reasoning
  }))
}

'use client'

import { useState } from 'react'
import { Sparkles, Loader2, User } from 'lucide-react'

export default function AIAgentSearch({ pazienti }: { pazienti: any[] }) {
  const [query, setQuery] = useState('')
  const [loading, setLoading] = useState(false)
  const [results, setResults] = useState<any[]>([])

  const handleSearch = async () => {
    if (!query.trim()) return
    
    setLoading(true)
    try {
      const res = await fetch('/api/ai-search', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query, pazienti })
      })
      
      const data = await res.json()
      setResults(data.results)
    } catch (error) {
      console.error('AI search error:', error)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="bg-gradient-to-r from-purple-50 via-blue-50 to-purple-50 rounded-xl border-2 border-purple-200 p-6">
      <div className="flex items-center gap-2 mb-4">
        <Sparkles className="w-6 h-6 text-purple-600" />
        <h2 className="text-xl font-bold text-gray-900">AI Agent Search</h2>
        <span className="px-2 py-1 bg-purple-100 text-purple-700 text-xs font-semibold rounded-full">
          PRO
        </span>
      </div>

      <p className="text-gray-600 mb-4">
        Ricerca semantica intelligente con comprensione del linguaggio naturale
      </p>

      <div className="flex gap-2">
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
          placeholder='Es: "paziente diabetico di Milano visto la settimana scorsa"'
          className="flex-1 px-4 py-3 border border-purple-300 rounded-lg focus:ring-2 focus:ring-purple-500 focus:border-transparent"
        />
        <button
          onClick={handleSearch}
          disabled={loading}
          className="px-6 py-3 bg-purple-600 text-white rounded-lg hover:bg-purple-700 disabled:opacity-50 flex items-center gap-2"
        >
          {loading ? (

              <Loader2 className="w-4 h-4 animate-spin" />
              Ricerca...
            </>
          ) : (
            <>
              <Sparkles className="w-4 h-4" />
              Cerca
            </>
          )}
        </button>
      </div>

      {results.length > 0 && (
        <div className="mt-6 space-y-3">
          {results.map((result, idx) => (
            <div key={idx} className="bg-white rounded-lg p-4 border border-purple-200">

import { checkFeatureAccess } from '@/lib/subscription'
import { aiAgentSearch } from '@/lib/ai-agent'

export async function POST(request: NextRequest) {
  try {
    // Check feature access
    const hasAccess = await checkFeatureAccess('ai-agent-search')
    
    if (!hasAccess) {
      return NextResponse.json(
        { error: 'Upgrade a Professional per usare AI Agent Search' },
        { status: 403 }
      )
    }

    const { query, pazienti } = await request.json()
    
    const results = await aiAgentSearch(query, pazienti)
    
    return NextResponse.json({ results })
  } catch (error) {
    console.error('AI search error:', error)
    return NextResponse.json({ error: 'Errore AI' }, { status: 500 })
  }
}
