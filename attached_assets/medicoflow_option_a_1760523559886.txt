// app/api/preferences/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { DEFAULT_PREFERENCES } from '@/types/preferences'

export async function GET(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    let preferences = await prisma.userPreferences.findUnique({
      where: { userId: session.user.id }
    })
    
    if (!preferences) {
      return NextResponse.json(DEFAULT_PREFERENCES)
    }
    
    return NextResponse.json({
      theme: preferences.theme,
      dashboardLayout: preferences.dashboardLayout,
      tableSettings: preferences.tableSettings,
      notifications: preferences.notifications,
      customShortcuts: preferences.customShortcuts
    })
    
  } catch (error) {
    console.error('Error fetching preferences:', error)
    return NextResponse.json(
      { error: 'Errore nel caricamento delle preferenze' }, 
      { status: 500 }
    )
  }
}

export async function PUT(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    const body = await request.json()
    
    if (body.theme && !['light', 'dark', 'auto'].includes(body.theme)) {
      return NextResponse.json(
        { error: 'Tema non valido' }, 
        { status: 400 }
      )
    }
    
    const preferences = await prisma.userPreferences.upsert({
      where: { userId: session.user.id },
      create: {
        userId: session.user.id,
        theme: body.theme || 'light',
        dashboardLayout: body.dashboardLayout || [],
        tableSettings: body.tableSettings || {},
        notifications: body.notifications || {},
        customShortcuts: body.customShortcuts
      },
      update: {
        ...(body.theme && { theme: body.theme }),
        ...(body.dashboardLayout && { dashboardLayout: body.dashboardLayout }),
        ...(body.tableSettings && { tableSettings: body.tableSettings }),
        ...(body.notifications && { notifications: body.notifications }),
        ...(body.customShortcuts !== undefined && { customShortcuts: body.customShortcuts })
      }
    })
    
    return NextResponse.json(preferences)
    
  } catch (error) {
    console.error('Error updating preferences:', error)
    return NextResponse.json(
      { error: 'Errore nel salvataggio delle preferenze' }, 
      { status: 500 }
    )
  }
}

export async function DELETE(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    await prisma.userPreferences.delete({
      where: { userId: session.user.id }
    })
    
    return NextResponse.json({ success: true })
    
  } catch (error) {
    console.error('Error deleting preferences:', error)
    return NextResponse.json(
      { error: 'Errore nel reset delle preferenze' }, 
      { status: 500 }
    )
  }
}

```typescript
// app/api/onboarding/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    let progress = await prisma.onboardingProgress.findUnique({
      where: { userId: session.user.id }
    })
    
    if (!progress) {
      progress = await prisma.onboardingProgress.create({
        data: { 
          userId: session.user.id,
          completedSteps: [],
          currentStep: 'welcome',
          isCompleted: false
        }
      })
    }
    
    return NextResponse.json(progress)
    
  } catch (error) {
    console.error('Error fetching onboarding progress:', error)
    return NextResponse.json(
      { error: 'Errore nel caricamento del progresso' }, 
      { status: 500 }
    )
  }
}

export async function POST(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    const { step, action } = await request.json()
    
    if (!['complete', 'skip'].includes(action)) {
      return NextResponse.json(
        { error: 'Azione non valida' }, 
        { status: 400 }
      )
    }
    
    const currentProgress = await prisma.onboardingProgress.findUnique({
      where: { userId: session.user.id }
    })
    
    if (!currentProgress) {
      return NextResponse.json(
        { error: 'Progresso onboarding non trovato' }, 
        { status: 404 }
      )
    }
    
    const stepOrder = ['welcome', 'ambulatorio', 'specializzazione', 'paziente', 'tutorial', 'complete']
    const currentIndex = stepOrder.indexOf(step)
    const nextStep = currentIndex < stepOrder.length - 1 ? stepOrder[currentIndex + 1] : null
    
    let updateData: any = {}
    
    if (action === 'complete') {
      const completedSteps = currentProgress.completedSteps as string[]
      if (!completedSteps.includes(step)) {
        completedSteps.push(step)
      }
      
      updateData = {
        completedSteps,
        currentStep: nextStep,
        isCompleted: nextStep === null,
        completedAt: nextStep === null ? new Date() : null
      }
    } else if (action === 'skip') {
      updateData = {
        isCompleted: true,
        skippedAt: new Date(),
        currentStep: null
      }
    }
    
    const progress = await prisma.onboardingProgress.update({
      where: { userId: session.user.id },
      data: updateData
    })
    
    return NextResponse.json(progress)
    
  } catch (error) {
    console.error('Error updating onboarding progress:', error)
    return NextResponse.json(
      { error: 'Errore nell\'aggiornamento del progresso' }, 
      { status: 500 }
    )
  }
}

// app/api/ambulatorio/specialization/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function PUT(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session?.user?.id) {
      return NextResponse.json(
        { error: 'Non autorizzato' }, 
        { status: 401 }
      )
    }
    
    // Ottieni ambulatorio dell'utente
    const user = await prisma.user.findUnique({
      where: { id: session.user.id },
      include: { ambulatorio: true }
    })
    
    if (!user?.ambulatorio) {
      return NextResponse.json(
        { error: 'Ambulatorio non trovato' }, 
        { status: 404 }
      )
    }
    
    const { specializationId } = await request.json()
    
    // Verifica che la specializzazione esista
    if (specializationId) {
      const specialization = await prisma.specialization.findUnique({
        where: { id: specializationId }
      })
      
      if (!specialization) {
        return NextResponse.json(
          { error: 'Specializzazione non valida' }, 
          { status: 400 }
        )
      }
    }
    
    // Aggiorna ambulatorio
    const updatedAmbulatorio = await prisma.ambulatorio.update({
      where: { id: user.ambulatorio.id },
      data: { specializationId },
      include: { specialization: true }
    })
    
    return NextResponse.json(updatedAmbulatorio)
    
  } catch (error) {
    console.error('Error updating ambulatorio specialization:', error)
    return NextResponse.json(
      { error: 'Errore nell\'aggiornamento della specializzazione' }, 
      { status: 500 }
    )
  }
}

// hooks/usePreferences.ts

import { useState, useEffect } from 'react'
import { UserPreferences, DEFAULT_PREFERENCES } from '@/types/preferences'

export function usePreferences() {
  const [preferences, setPreferences] = useState<UserPreferences | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  
  useEffect(() => {
    fetchPreferences()
  }, [])
  
  const fetchPreferences = async () => {
    try {
      setLoading(true)
      const res = await fetch('/api/preferences')
      
      if (!res.ok) {
        throw new Error('Errore nel caricamento delle preferenze')
      }
      
      const data = await res.json()
      setPreferences(data)
      applyTheme(data.theme)
      
    } catch (err) {
      console.error('Error fetching preferences:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
      setPreferences(DEFAULT_PREFERENCES)
    } finally {
      setLoading(false)
    }
  }
  
  const updatePreferences = async (updates: Partial<UserPreferences>) => {
    try {
      const res = await fetch('/api/preferences', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updates)
      })
      
      if (!res.ok) {
        throw new Error('Errore nel salvataggio delle preferenze')
      }
      
      const data = await res.json()
      setPreferences(prev => ({ ...prev, ...data }))
      
      if (updates.theme) {
        applyTheme(updates.theme)
      }
      
      return data
    } catch (err) {
      console.error('Error updating preferences:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
      throw err
    }
  }
  
  const resetPreferences = async () => {
    try {
      await fetch('/api/preferences', { method: 'DELETE' })
      setPreferences(DEFAULT_PREFERENCES)
      applyTheme('light')
    } catch (err) {
      console.error('Error resetting preferences:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
      throw err
    }
  }
  
  const applyTheme = (theme: string) => {
    if (theme === 'dark') {
      document.documentElement.classList.add('dark')
    } else if (theme === 'light') {
      document.documentElement.classList.remove('dark')
    } else if (theme === 'auto') {
      const isDark = window.matchMedia('(prefers-color-scheme: dark)').matches
      if (isDark) {
        document.documentElement.classList.add('dark')
      } else {
        document.documentElement.classList.remove('dark')
      }
    }
  }
  
  return {
    preferences,
    loading,
    error,
    updatePreferences,
    resetPreferences,
    refetch: fetchPreferences
  }
}
```

// hooks/useOnboarding.ts

import { useState, useEffect } from 'react'
import { OnboardingProgress } from '@/types/preferences'

export function useOnboarding() {
  const [progress, setProgress] = useState<OnboardingProgress | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  
  useEffect(() => {
    fetchProgress()
  }, [])
  
  const fetchProgress = async () => {
    try {
      setLoading(true)
      const res = await fetch('/api/onboarding')
      
      if (!res.ok) {
        throw new Error('Errore nel caricamento del progresso')
      }
      
      const data = await res.json()
      setProgress(data)
    } catch (err) {
      console.error('Error fetching onboarding progress:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
    } finally {
      setLoading(false)
    }
  }
  
  const completeStep = async (step: string) => {
    try {
      const res = await fetch('/api/onboarding', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ step, action: 'complete' })
      })
      
      if (!res.ok) {
        throw new Error('Errore nell\'aggiornamento del progresso')
      }
      
      const data = await res.json()
      setProgress(data)
      return data
    } catch (err) {
      console.error('Error completing step:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
      throw err
    }
  }
  
  const skipOnboarding = async () => {
    try {
      const res = await fetch('/api/onboarding', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ step: 'skip', action: 'skip' })
      })
      
      if (!res.ok) {
        throw new Error('Errore nel skip dell\'onboarding')
      }
      
      const data = await res.json()
      setProgress(data)
      return data
    } catch (err) {
      console.error('Error skipping onboarding:', err)
      setError(err instanceof Error ? err.message : 'Errore sconosciuto')
      throw err
    }
  }
  
  return {
    progress,
    loading,
    error,
    isCompleted: progress?.isCompleted || false,
    currentStep: progress?.currentStep || null,
    completedSteps: (progress?.completedSteps as string[]) || [],
    completeStep,
    skipOnboarding,
    refetch: fetchProgress
  }
}

// hooks/useMediaQuery.ts

import { useState, useEffect } from 'react'

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(false)
  
  useEffect(() => {
    const media = window.matchMedia(query)
    setMatches(media.matches)
    
    const listener = (event: MediaQueryListEvent) => {
      setMatches(event.matches)
    }
    
    media.addEventListener('change', listener)
    return () => media.removeEventListener('change', listener)
  }, [query])
  
  return matches
}

export function useIsMobile() {
  return useMediaQuery('(max-width: 768px)')
}

export function useIsTablet() {
  return useMediaQuery('(min-width: 769px) and (max-width: 1024px)')
}

export function useIsDesktop() {
  return useMediaQuery('(min-width: 1025px)')
}

  { id: 'tutorial', title: 'Tutorial', component: TutorialStep }
]

export default function OnboardingWizard() {
  const router = useRouter()
  const { progress, completeStep, skipOnboarding } = useOnboarding()
  const [currentStepIndex, setCurrentStepIndex] = useState(0)
  
  const currentStep = STEPS[currentStepIndex]
  const StepComponent = currentStep.component
  const progressPercent = ((currentStepIndex + 1) / STEPS.length) * 100
  
  const handleNext = async () => {
    await completeStep(currentStep.id)
    
    if (currentStepIndex < STEPS.length - 1) {
      setCurrentStepIndex(currentStepIndex + 1)
    } else {
      router.push('/dashboard')
    }
  }
  
  const handleSkip = async () => {
    await skipOnboarding()
    router.push('/dashboard')
  }
  
  const handleBack = () => {
    if (currentStepIndex > 0) {
      setCurrentStepIndex(currentStepIndex - 1)
    }
  }
  
  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-100 dark:from-gray-900 dark:to-gray-800 flex items-center justify-center p-4">
      <Card className="w-full max-w-2xl">
        <CardHeader>
          <div className="mb-4">
            <Progress value={progressPercent} className="h-2" />
          </div>
          <div className="flex items-center justify-between">
            <div>
              <CardTitle className="text-2xl">{currentStep.title}</CardTitle>
              <CardDescription>
                Passo {currentStepIndex + 1} di {STEPS.length}
              </CardDescription>
            </div>
          </div>
        </CardHeader>
        
        <CardContent className="space-y-6">
          <StepComponent onNext={handleNext} />
          
          <div className="flex justify-between pt-6 border-t">
            <div>
              {currentStepIndex > 0 && (
                <Button variant="outline" onClick={handleBack}>
                  Indietro
                </Button>
              )}
            </div>
            
            <div className="flex gap-2">
              <Button variant="ghost" onClick={handleSkip}>
                Salta configurazione
              </Button>
              <Button onClick={handleNext}>
                {currentStepIndex === STEPS.length - 1 ? '🎉 Vai alla Dashboard' : 'Avanti'}
              </Button>
            </div>
          </div>
        </CardContent>
      </Card>
    </div>
  )
}

'use client'

import { useState, useEffect } from 'react'
import { useSession } from 'next-auth/react'
import { useSpecializations } from '@/hooks/useSpecializations'
import { Label } from '@/components/ui/label'
import { Input } from '@/components/ui/input'
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
import { Button } from '@/components/ui/button'
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
import { Loader2, Building2 } from 'lucide-react'
import { useToast } from '@/hooks/use-toast'

export default function AmbulatorioSettings() {
  const { data: session } = useSession()
  const { specializations, loading: loadingSpecs } = useSpecializations()
  const { toast } = useToast()
  const [loading, setLoading] = useState(true)
  const [saving, setSaving] = useState(false)
  
  const [formData, setFormData] = useState({
    nome: '',
    indirizzo: '',
    citta: '',
    telefono: '',
    specializationId: ''
  })
  
  useEffect(() => {
    fetchAmbulatorio()
  }, [])
  
  const fetchAmbulatorio = async () => {
    try {
      const res = await fetch('/api/ambulatorio')
      const data = await res.json()
      
      setFormData({
        nome: data.nome || '',
        indirizzo: data.indirizzo || '',
        citta: data.citta || '',
        telefono: data.telefono || '',
        specializationId: data.specializationId || ''
      })
    } catch (error) {
      console.error('Error fetching ambulatorio:', error)
    } finally {
      setLoading(false)
    }
  }
  
  const handleSave = async () => {
    setSaving(true)
    try {
      // Salva dati ambulatorio
      await fetch('/api/ambulatorio', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData)

        await fetch('/api/ambulatorio/specialization', {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },

        })

        title: 'Salvato!',
        description: 'I dati dell\'ambulatorio sono stati aggiornati.'
      })

      toast({
        title: 'Errore',
        description: 'Si è verificato un errore durante il salvataggio.',
        variant: 'destructive'
      })
    } finally {
      setSaving(false)
    }
  }
  
  if (loading || loadingSpecs) {
    return (
      <div className="flex justify-center py-12">
        <Loader2 className="w-8 h-8 animate-spin" />
      </div>
    )
  }
  
  return (
    <Card>
      <CardHeader>
        <div className="flex items-center gap-2">
          <Building2 className="w-5 h-5" />
          <CardTitle>Dati Ambulatorio</CardTitle>
        </div>
        <CardDescription>
          Gestisci le informazioni del tuo ambulatorio
        </CardDescription>
      </CardHeader>
      <CardContent className="space-y-4">
        <div className="space-y-2">
          <Label htmlFor="nome">Nome Ambulatorio</Label>
          <Input
            id="nome"
            value={formData.nome}
            onChange={(e) => setFormData({ ...formData, nome: e.target.value })}
            placeholder="Studio Medico Rossi"
          />
        </div>
        
        <div className="space-y-2">
          <Label htmlFor="indirizzo">Indirizzo</Label>
          <Input
            id="indirizzo"
            value={formData.indirizzo}
            onChange={(e) => setFormData({ ...formData, indirizzo: e.target.value })}
            placeholder="Via Roma, 123"
          />
        </div>
        
        <div className="grid grid-cols-2 gap-4">
          <div className="space-y-2">
            <Label htmlFor="citta">Città</Label>
            <Input
              id="citta"
              value={formData.citta}
              onChange={(e) => setFormData({ ...formData, citta: e.target.value })}
