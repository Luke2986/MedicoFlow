# Prompt Replit: Implementa Sistema Smoke Test Pricing con Lead Capture

Implementa un sistema completo di smoke test per validare i prezzi senza integrazione Stripe reale. Il sistema deve raccogliere lead qualificati quando gli utenti mostrano interesse per un piano tariffario.

## OBIETTIVO
Creare pagina pricing con 3 piani (Base €89, Professional €199, Enterprise €399) che raccoglie lead invece di processare pagamenti. Include dashboard admin per analizzare l'interesse.

## STEP 1: Aggiorna Prisma Schema

Aggiungi questo model alla fine di `prisma/schema.prisma`:

```prisma
model PricingLead {
  id            String   @id @default(cuid())
  email         String
  nome          String
  ambulatorio   String?
  piano         String   // BASE, PROFESSIONAL, ENTERPRISE
  telefono      String?
  note          String?
  status        String   @default("nuovo") // nuovo, contattato, convertito
  createdAt     DateTime @default(now())
  
  @@index([piano])
  @@index([status])
  @@index([createdAt])
}
```

Esegui migrazione:
```bash
npx prisma migrate dev --name add_pricing_leads
npx prisma generate
```

## STEP 2: Crea API Route per Lead

Crea il file `app/api/pricing-leads/route.ts`:

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function POST(request: NextRequest) {
  try {
    const body = await request.json()
    
    const lead = await prisma.pricingLead.create({
      data: {
        nome: body.nome,
        email: body.email,
        ambulatorio: body.ambulatorio || null,
        telefono: body.telefono || null,
        piano: body.piano,
        note: body.note || null,
        status: 'nuovo'
      }
    })

    console.log(`🎯 Nuovo lead piano ${body.piano}:`, body.email)

    return NextResponse.json({ success: true, leadId: lead.id }, { status: 201 })
  } catch (error) {
    console.error('Errore creazione lead:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function GET() {
  try {
    const leads = await prisma.pricingLead.findMany({
      orderBy: { createdAt: 'desc' }
    })

    const stats = {
      total: leads.length,
      byPlan: {
        BASE: leads.filter(l => l.piano === 'BASE').length,
        PROFESSIONAL: leads.filter(l => l.piano === 'PROFESSIONAL').length,
        ENTERPRISE: leads.filter(l => l.piano === 'ENTERPRISE').length,
      },
      byStatus: {
        nuovo: leads.filter(l => l.status === 'nuovo').length,
        contattato: leads.filter(l => l.status === 'contattato').length,
        convertito: leads.filter(l => l.status === 'convertito').length,
      }
    }

    return NextResponse.json({ leads, stats })
  } catch (error) {
    console.error('Errore fetch leads:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

function StatCard({ 
  label, 
  value, 
  percentage,
  icon: Icon, 
  color = 'blue' 
}: { 
  label: string
  value: number
  percentage?: number
  icon: any
  color?: 'blue' | 'purple' | 'gray'
}) {
  const colors = {
    blue: 'bg-blue-100 text-blue-600',
    purple: 'bg-purple-100 text-purple-600',
    gray: 'bg-gray-100 text-gray-600'
  }

  return (
    <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
      <div className="flex items-center justify-between mb-4">
        <div className={`p-3 rounded-lg ${colors[color]}`}>
          <Icon className="w-6 h-6" />
        </div>
        {percentage !== undefined && (
          <span className="text-sm font-semibold text-gray-600">
            {percentage}%
          </span>
        )}
      </div>
      <p className="text-sm text-gray-600 mb-1">{label}</p>
      <p className="text-3xl font-bold text-gray-900">{value}</p>
    </div>
  )
}
