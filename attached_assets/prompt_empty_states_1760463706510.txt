```prisma
model OnboardingProgress {
  id              String   @id @default(cuid())
  ambulatorioId   String   @unique
  
  // Step completati
  profileCompleted      Boolean @default(false)
  firstPatientAdded     Boolean @default(false)
  firstAppointmentAdded Boolean @default(false)
  themeCustomized       Boolean @default(false)
  
  // Metadata
  onboardingCompleted   Boolean @default(false)
  completedAt           DateTime?
  
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@index([ambulatorioId])
}
```

## 📄 FILE 2: app/api/onboarding/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'

export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    // Get o crea progress
    let progress = await prisma.onboardingProgress.findUnique({
      where: { ambulatorioId: user.ambulatorioId }
    })

    if (!progress) {
      progress = await prisma.onboardingProgress.create({
        data: { ambulatorioId: user.ambulatorioId }
      })
    }

    // Check stato reale
    const [ambulatorio, pazientiCount, appuntamentiCount, themeSettings] = await Promise.all([
      prisma.ambulatorio.findUnique({
        where: { id: user.ambulatorioId }
      }),
      prisma.paziente.count({
        where: { ambulatorioId: user.ambulatorioId }
      }),
      prisma.appuntamento.count({
        where: { ambulatorioId: user.ambulatorioId }
      }),
      prisma.themeSettings.findUnique({
        where: { ambulatorioId: user.ambulatorioId }
      })
    ])

    // Auto-update progress
    const updates: any = {}
    let needsUpdate = false

    if (!progress.profileCompleted && ambulatorio?.telefono && ambulatorio?.piva) {
      updates.profileCompleted = true
      needsUpdate = true
    }

    if (!progress.firstPatientAdded && pazientiCount > 0) {
      updates.firstPatientAdded = true
      needsUpdate = true
    }

    if (!progress.firstAppointmentAdded && appuntamentiCount > 0) {
      updates.firstAppointmentAdded = true
      needsUpdate = true
    }

    if (!progress.themeCustomized && themeSettings) {
      updates.themeCustomized = true
      needsUpdate = true
    }

    // Check se tutto completato
    const allCompleted = 
      (progress.profileCompleted || updates.profileCompleted) &&
      (progress.firstPatientAdded || updates.firstPatientAdded) &&
      (progress.firstAppointmentAdded || updates.firstAppointmentAdded) &&
      (progress.themeCustomized || updates.themeCustomized)

    if (allCompleted && !progress.onboardingCompleted) {
      updates.onboardingCompleted = true
      updates.completedAt = new Date()
      needsUpdate = true
    }

    if (needsUpdate) {
      progress = await prisma.onboardingProgress.update({
        where: { ambulatorioId: user.ambulatorioId },
        data: updates
      })
    }

    return NextResponse.json({
      progress,
      stats: {
        pazientiCount,
        appuntamentiCount,
        hasTheme: !!themeSettings
      }
    })
  } catch (error) {
    console.error('Errore onboarding:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const body = await request.json()
    const { step } = body

    const updates: any = {}
    if (step === 'profile') updates.profileCompleted = true
    if (step === 'patient') updates.firstPatientAdded = true
    if (step === 'appointment') updates.firstAppointmentAdded = true
    if (step === 'theme') updates.themeCustomized = true

    const progress = await prisma.onboardingProgress.upsert({
      where: { ambulatorioId: user.ambulatorioId },
      create: {
        ambulatorioId: user.ambulatorioId,
        ...updates
      },
      update: updates
    })

    return NextResponse.json(progress)
  } catch (error) {
    console.error('Errore update onboarding:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

## 📄 FILE 4: components/dashboard/OnboardingWizard.tsx

```typescript
'use client'

import { useState, useEffect } from 'react'
import { X, Check, ChevronRight, User, Users, Calendar, Palette } from 'lucide-react'
import Link from 'next/link'

export default function OnboardingWizard() {
  const [show, setShow] = useState(false)
  const [progress, setProgress] = useState<any>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetchProgress()
  }, [])

  const fetchProgress = async () => {
    try {
      const res = await fetch('/api/onboarding')
      const data = await res.json()
      setProgress(data.progress)
      
      // Mostra wizard solo se non completato
      if (!data.progress.onboardingCompleted) {
        setShow(true)
      }
    } catch (error) {
      console.error('Errore fetch onboarding:', error)
    } finally {
      setLoading(false)
    }
  }

  const handleDismiss = () => {
    setShow(false)
    // Salva in localStorage per non mostrarlo più in questa sessione
    localStorage.setItem('onboarding_dismissed', 'true')
  }

  if (loading || !show || !progress) return null

  const steps = [
    {
      id: 'profile',
      title: 'Completa il Profilo',
      description: 'Aggiungi i dati del tuo ambulatorio',
      icon: User,
      href: '/dashboard/impostazioni/ambulatorio',
      completed: progress.profileCompleted
    },
    {
      id: 'patient',
      title: 'Aggiungi un Paziente',
      description: 'Inserisci il tuo primo paziente',
      icon: Users,
      href: '/dashboard/pazienti/nuovo',
      completed: progress.firstPatientAdded
    },
    {
      id: 'appointment',
      title: 'Crea un Appuntamento',
      description: 'Pianifica la prima visita',
      icon: Calendar,
      href: '/dashboard/appuntamenti',
      completed: progress.firstAppointmentAdded
    },
    {
      id: 'theme',
      title: 'Personalizza Aspetto',
      description: 'Configura i colori del brand',
      icon: Palette,
      href: '/dashboard/impostazioni/aspetto',
      completed: progress.themeCustomized
    }
  ]

  const completedSteps = steps.filter(s => s.completed).length
  const progressPercent = (completedSteps / steps.length) * 100

  return (
    <div className="bg-gradient-to-br from-blue-50 to-blue-100 rounded-xl border-2 border-blue-200 p-6 relative overflow-hidden">
      <button
        onClick={handleDismiss}
        className="absolute top-4 right-4 p-1 hover:bg-blue-200 rounded-lg transition-colors"
      >
        <X className="w-5 h-5 text-blue-600" />
      </button>

      <div className="mb-6">
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          Benvenuto su MedicoFlow! 🎉
        </h2>
        <p className="text-gray-700">
          Completa questi passaggi per configurare il tuo ambulatorio
        </p>
      </div>

      {/* Progress Bar */}
      <div className="mb-6">
        <div className="flex items-center justify-between mb-2">
          <span className="text-sm font-medium text-gray-700">
            Progresso: {completedSteps}/{steps.length}
          </span>
          <span className="text-sm font-semibold text-blue-600">
            {Math.round(progressPercent)}%
          </span>
        </div>
        <div className="w-full bg-blue-200 rounded-full h-3">
          <div
            className="bg-blue-600 h-3 rounded-full transition-all duration-500"
            style={{ width: `${progressPercent}%` }}
          />
        </div>
      </div>

      {/* Steps */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        {steps.map((step) => {
          const Icon = step.icon
          return (
            <Link
              key={step.id}
              href={step.href}
              className={`flex items-start gap-4 p-4 rounded-lg border-2 transition-all ${
                step.completed
                  ? 'bg-green-50 border-green-200'
                  : 'bg-white border-blue-200 hover:border-blue-400 hover:shadow-md'
              }`}
            >
              <div className={`p-3 rounded-lg ${
                step.completed ? 'bg-green-100' : 'bg-blue-100'
              }`}>
                {step.completed ? (
                  <Check className="w-6 h-6 text-green-600" />
                ) : (
                  <Icon className="w-6 h-6 text-blue-600" />
                )}
              </div>
              <div className="flex-1">
                <h3 className="font-semibold text-gray-900 mb-1">
                  {step.title}
                </h3>
                <p className="text-sm text-gray-600">
                  {step.description}
                </p>
              </div>
              {!step.completed && (
                <ChevronRight className="w-5 h-5 text-gray-400 mt-1" />
              )}
            </Link>
          )
        })}
      </div>

      {completedSteps === steps.length && (
        <div className="mt-6 p-4 bg-green-100 border-2 border-green-300 rounded-lg text-center">
          <p className="text-green-800 font-semibold">
            🎊 Congratulazioni! Hai completato la configurazione!
          </p>
        </div>
      )}
    </div>
  )
}
```

## 📄 FILE 5: components/dashboard/QuickActions.tsx

```typescript
'use client'

import { Plus, Calendar, FileText, Users } from 'lucide-react'
import Link from 'next/link'

interface QuickActionsProps {
  hasPazienti: boolean
  hasAppuntamenti: boolean
}

export default function QuickActions({ hasPazienti, hasAppuntamenti }: QuickActionsProps) {
  const actions = [
    {
      icon: Users,
      label: 'Nuovo Paziente',
      href: '/dashboard/pazienti/nuovo',
      color: 'blue',
      priority: !hasPazienti
    },
    {
      icon: Calendar,
      label: 'Nuovo Appuntamento',
      href: '/dashboard/appuntamenti',
      color: 'green',
      priority: hasPazienti && !hasAppuntamenti
    },
    {
      icon: FileText,
      label: 'Nuova Cartella',
      href: '/dashboard/cartella-clinica/nuovo',
      color: 'purple',
      priority: false,
      disabled: !hasPazienti
    }
  ]

  const priorityActions = actions.filter(a => a.priority)
  const normalActions = actions.filter(a => !a.priority)

  const colorClasses = {
    blue: 'bg-blue-50 border-blue-200 text-blue-700 hover:bg-blue-100',
    green: 'bg-green-50 border-green-200 text-green-700 hover:bg-green-100',
    purple: 'bg-purple-50 border-purple-200 text-purple-700 hover:bg-purple-100'
  }

  return (
    <div className="space-y-4">
      {priorityActions.length > 0 && (
        <div>
          <h3 className="text-sm font-semibold text-gray-700 mb-3">
            ⚡ Azioni Suggerite
          </h3>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
            {priorityActions.map((action) => {
              const Icon = action.icon
              return (
                <Link
                  key={action.label}
                  href={action.href}
                  className={`flex items-center gap-3 p-4 border-2 rounded-lg transition-all ${
                    colorClasses[action.color as keyof typeof colorClasses]
                  }`}
                >
                  <div className="p-2 bg-white rounded-lg">
                    <Icon className="w-5 h-5" />
                  </div>
                  <span className="font-medium">{action.label}</span>
                </Link>
              )
            })}
          </div>
        </div>
      )}

      <div>
        <h3 className="text-sm font-semibold text-gray-700 mb-3">
          Quick Actions
        </h3>
        <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
          {normalActions.map((action) => {
            const Icon = action.icon
            return action.disabled ? (
              <div
                key={action.label}
                className="flex items-center gap-3 p-4 border-2 border-gray-200 rounded-lg opacity-50 cursor-not-allowed"
              >
                <Icon className="w-5 h-5 text-gray-400" />
                <span className="text-gray-500">{action.label}</span>
              </div>
            ) : (
              <Link
                key={action.label}
                href={action.href}
                className={`flex items-center gap-3 p-4 border-2 rounded-lg transition-all ${
                  colorClasses[action.color as keyof typeof colorClasses]
                }`}
              >
                <Icon className="w-5 h-5" />
                <span className="font-medium">{action.label}</span>
              </Link>
            )
          })}
        </div>
      </div>
    </div>
  )
}
```

## 📄 FILE 6: Aggiorna app/dashboard/page.tsx

```typescript
import { Users, Calendar, TrendingUp, AlertTriangle } from 'lucide-react'
import StatCard from '@/components/dashboard/StatCard'
import AppointmentTimeline from '@/components/dashboard/AppointmentTimeline'
import OnboardingWizard from '@/components/dashboard/OnboardingWizard'
import QuickActions from '@/components/dashboard/QuickActions'
import EmptyState from '@/components/dashboard/EmptyState'
import Link from 'next/link'
import { getCurrentUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

async function getDashboardData(ambulatorioId: string) {
  const [stats, appuntamentiOggi] = await Promise.all([
    prisma.$transaction([
      prisma.paziente.count({ where: { ambulatorioId } }),
      prisma.appuntamento.count({
        where: {
          ambulatorioId,
          dataOra: {
            gte: new Date(new Date().setHours(0, 0, 0, 0)),
            lte: new Date(new Date().setHours(23, 59, 59, 999))
          }
        }
      }),
      prisma.appuntamento.count({
        where: {
          ambulatorioId,
          dataOra: {
            gte: new Date(new Date().setDate(new Date().getDate() - 7))
          }
        }
      })
    ]),
    prisma.appuntamento.findMany({
      where: {
        ambulatorioId,
        dataOra: {
          gte: new Date(new Date().setHours(0, 0, 0, 0)),
          lte: new Date(new Date().setHours(23, 59, 59, 999))
        }
      },
      include: {
        paziente: {
          select: { nome: true, cognome: true }
        }
      },
      orderBy: { dataOra: 'asc' }
    })
  ])

  return {
    totalePazienti: stats[0],
    appuntamentiOggi: stats[1],
    appuntamentiSettimana: stats[2],
    prossimiAppuntamenti: appuntamentiOggi,
    hasPazienti: stats[0] > 0,
    hasAppuntamenti: stats[1] > 0 || stats[2] > 0
  }
}

export default async function DashboardPage() {
  const user = await getCurrentUser()
  if (!user) return null

  const data = await getDashboardData(user.ambulatorioId)

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Dashboard</h1>
          <p className="text-gray-600 mt-1">Panoramica del tuo ambulatorio</p>
        </div>
      </div>

      {/* Onboarding Wizard (mostra se non completato) */}
      <OnboardingWizard />

      {/* Stats Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
        <StatCard
          title="Pazienti Totali"
          value={data.totalePazienti}
          icon={Users}
          color="blue"
        />
        <StatCard
          title="Appuntamenti Oggi"
          value={data.appuntamentiOggi}
          icon={Calendar}
          color="green"
        />
        <StatCard
          title="Appuntamenti Settimana"
          value={data.appuntamentiSettimana}
          icon={TrendingUp}
          color="yellow"
        />
        <StatCard
          title="Tasso Attività"
          value={data.hasAppuntamenti ? "Alta" : "Bassa"}
          icon={AlertTriangle}
          color={data.hasAppuntamenti ? "green" : "red"}
        />
      </div>

      {/* Quick Actions */}
      <QuickActions 
        hasPazienti={data.hasPazienti}
        hasAppuntamenti={data.hasAppuntamenti}
      />

      {/* Appuntamenti Oggi */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
        <h2 className="text-xl font-semibold text-gray-900 mb-4">
          Appuntamenti di Oggi
        </h2>
        {data.prossimiAppuntamenti.length === 0 ? (
          <EmptyState
            icon={Calendar}
            title="Nessun appuntamento oggi"
            description="Inizia a pianificare le visite per i tuoi pazienti"
            actionLabel="Crea Appuntamento"
            actionHref="/dashboard/appuntamenti"
            secondaryAction={{
              label: "Vedi Calendario",
              href: "/dashboard/appuntamenti"
            }}
          />
        ) : (
          <AppointmentTimeline appointments={data.prossimiAppuntamenti} />
        )}
      </div>

      {/* Empty state pazienti */}
      {!data.hasPazienti && (
        <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-6">
          <EmptyState
            icon={Users}
            title="Aggiungi il tuo primo paziente"
            description="Inizia a gestire le cartelle cliniche dei tuoi pazienti in modo digitale e organizzato"
            actionLabel="Aggiungi Paziente"
            actionHref="/dashboard/pazienti/nuovo"
            secondaryAction={{
              label: "Importa da CSV",
              href: "/dashboard/pazienti"
            }}
          />
        </div>
      )}
    </div>
  )
}
```
