# 📄 STEP 3: Export PDF Professionale Cartelle Cliniche

## OBIETTIVO
Implementare generazione PDF professionale delle cartelle cliniche con intestazione ambulatorio, firma digitale medico e conformità medico-legale.

**Features:**
- PDF con logo e intestazione ambulatorio
- Sezioni strutturate (anamnesi, esame, diagnosi, terapia)
- Footer con firma medico e timbro
- Watermark "Copia conforme all'originale digitale"
- Metadati PDF (autore, data, titolo)
- Download diretto o preview browser

## 📄 FILE 2: app/api/cartelle/[id]/pdf/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { renderToBuffer } from '@react-pdf/renderer'
import CartellaClinicaPDF from '@/lib/pdf/cartella-pdf'
import { format } from 'date-fns'

export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    // Fetch cartella con tutte le relazioni
    const cartella = await prisma.cartellaClinica.findFirst({
      where: {
        id: params.id,
        paziente: {
          ambulatorioId: user.ambulatorioId
        }
      },
      include: {
        paziente: {
          select: {
            id: true,
            nome: true,
            cognome: true,
            codiceFiscale: true,
            dataNascita: true
          }
        }
      }
    })

    if (!cartella) {
      return NextResponse.json({ error: 'Cartella non trovata' }, { status: 404 })
    }

    // Fetch ambulatorio
    const ambulatorio = await prisma.ambulatorio.findUnique({
      where: { id: user.ambulatorioId }
    })

    if (!ambulatorio) {
      return NextResponse.json({ error: 'Ambulatorio non trovato' }, { status: 404 })
    }

    // Genera PDF
    const pdfDoc = CartellaClinicaPDF({
      cartella,
      ambulatorio,
      medico: { name: user.name }
    })

    const pdfBuffer = await renderToBuffer(pdfDoc)

    // Genera nome file
    const filename = `Cartella_${cartella.paziente.cognome}_${cartella.paziente.nome}_${format(new Date(cartella.dataVisita), 'yyyy-MM-dd')}.pdf`

    return new NextResponse(pdfBuffer, {
      headers: {
        'Content-Type': 'application/pdf',
        'Content-Disposition': `attachment; filename="${filename}"`,
        'Cache-Control': 'no-cache'
      }
    })
  } catch (error) {
    console.error('Errore generazione PDF:', error)
    return NextResponse.json({ error: 'Errore generazione PDF' }, { status: 500 })
  }
}
```

// Nel componente CartellaView, aggiungi questo button nella sezione header actions:

const [downloading, setDownloading] = useState(false)

const handleExportPDF = async () => {
  setDownloading(true)
  try {
    const res = await fetch(`/api/cartelle/${cartella.id}/pdf`)
    if (!res.ok) throw new Error('Errore download')
    
    const blob = await res.blob()
    const url = window.URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `Cartella_${cartella.paziente.cognome}_${cartella.paziente.nome}.pdf`
    document.body.appendChild(a)
    a.click()
    window.URL.revokeObjectURL(url)
    document.body.removeChild(a)
  } catch (error) {
    alert('Errore durante il download del PDF')
  } finally {
    setDownloading(false)
  }
}

// Nel JSX, sostituisci il button "Export PDF" con:
<button
  onClick={handleExportPDF}
  disabled={downloading}

    <>
      <Loader2 className="w-4 h-4 animate-spin" />
      Generazione PDF...
    </>
  ) : (
    <>
      <Download className="w-4 h-4" />
