```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'

export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const searchParams = request.nextUrl.searchParams
    
    // Search query
    const q = searchParams.get('q') || ''
    
    // Filtri
    const sesso = searchParams.get('sesso') || 'all'
    const citta = searchParams.get('citta') || 'all'
    const etaMin = parseInt(searchParams.get('etaMin') || '0')
    const etaMax = parseInt(searchParams.get('etaMax') || '120')
    const ultimaVisita = searchParams.get('ultimaVisita') || 'all' // all, 30, 90, 180, 365
    
    // Paginazione
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const skip = (page - 1) * limit
    
    // Ordinamento
    const sortBy = searchParams.get('sortBy') || 'cognome'
    const sortOrder = searchParams.get('sortOrder') || 'asc'

    // Build where clause
    const where: any = {
      ambulatorioId: user.ambulatorioId
    }

    // Search
    if (q) {
      where.OR = [
        { nome: { contains: q, mode: 'insensitive' } },
        { cognome: { contains: q, mode: 'insensitive' } },
        { codiceFiscale: { contains: q, mode: 'insensitive' } },
        { telefono: { contains: q, mode: 'insensitive' } },
        { email: { contains: q, mode: 'insensitive' } }
      ]
    }

    // Filtro sesso
    if (sesso !== 'all') {
      where.sesso = sesso
    }

    // Filtro città
    if (citta !== 'all') {
      where.citta = citta
    }

    // Filtro età (calcolo anno nascita)
    const currentYear = new Date().getFullYear()
    if (etaMin > 0 || etaMax < 120) {
      where.dataNascita = {
        gte: new Date(currentYear - etaMax, 0, 1),
        lte: new Date(currentYear - etaMin, 11, 31)
      }
    }

    // Query pazienti
    const [pazienti, total] = await Promise.all([
      prisma.paziente.findMany({
        where,
        include: {
          appuntamenti: {
            orderBy: { dataOra: 'desc' },
            take: 1,
            select: { dataOra: true, stato: true }
          },
          cartelle: {
            orderBy: { dataVisita: 'desc' },
            take: 1,
            select: { dataVisita: true }
          }
        },
        orderBy: { [sortBy]: sortOrder },
        skip,
        take: limit
      }),
      prisma.paziente.count({ where })
    ])

    // Calcola età e status per ciascun paziente
    const pazientiEnriched = pazienti.map(p => {
      const dataNascita = new Date(p.dataNascita)
      const eta = currentYear - dataNascita.getFullYear()
      
      const ultimoAppuntamento = p.appuntamenti[0]
      const ultimaCartella = p.cartelle[0]
      
      // Determina status
      let status = 'inattivo'
      const giorniDaUltimaVisita = ultimaCartella 
        ? Math.floor((Date.now() - new Date(ultimaCartella.dataVisita).getTime()) / (1000 * 60 * 60 * 24))
        : 999

      if (giorniDaUltimaVisita < 30) status = 'attivo'
      else if (giorniDaUltimaVisita < 90) status = 'recente'
      
      // È nuovo se creato negli ultimi 7 giorni
      const isNuovo = (Date.now() - new Date(p.createdAt).getTime()) < 7 * 24 * 60 * 60 * 1000

      return {
        ...p,
        eta,
        status,
        isNuovo,
        ultimoAppuntamento: ultimoAppuntamento?.dataOra,
        ultimaCartella: ultimaCartella?.dataVisita
      }
    })

    // Filtro ultima visita
    let pazientiFiltrati = pazientiEnriched
    if (ultimaVisita !== 'all') {
      const giorni = parseInt(ultimaVisita)
      const cutoffDate = new Date()
      cutoffDate.setDate(cutoffDate.getDate() - giorni)
      
      pazientiFiltrati = pazientiEnriched.filter(p => 
        p.ultimaCartella && new Date(p.ultimaCartella) >= cutoffDate
      )
    }

    return NextResponse.json({
      pazienti: pazientiFiltrati,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit)
      }
    })
  } catch (error) {
    console.error('Errore search pazienti:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

## 📄 FILE 2: app/api/pazienti/export/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { format } from 'date-fns'

export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const pazienti = await prisma.paziente.findMany({
      where: { ambulatorioId: user.ambulatorioId },
      orderBy: { cognome: 'asc' }
    })

    // Genera CSV
    const headers = [
      'Nome',
      'Cognome',
      'Codice Fiscale',
      'Data Nascita',
      'Sesso',
      'Indirizzo',
      'Città',
      'Provincia',
      'CAP',
      'Telefono',
      'Email',
      'Note'
    ]

    const rows = pazienti.map(p => [
      p.nome,
      p.cognome,
      p.codiceFiscale,
      format(new Date(p.dataNascita), 'dd/MM/yyyy'),
      p.sesso,
      p.indirizzo,
      p.citta,
      p.provincia,
      p.cap,
      p.telefono,
      p.email || '',
      p.note || ''
    ])

    const csv = [
      headers.join(','),
      ...rows.map(row => row.map(cell => `"${cell}"`).join(','))
    ].join('\n')

    return new NextResponse(csv, {
      headers: {
        'Content-Type': 'text/csv',
        'Content-Disposition': `attachment; filename="pazienti-${format(new Date(), 'yyyy-MM-dd')}.csv"`
      }
    })
  } catch (error) {
    console.error('Errore export CSV:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

## 📄 FILE 3: app/api/pazienti/import/route.ts

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import Papa from 'papaparse'

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const formData = await request.formData()
    const file = formData.get('file') as File

    if (!file) {
      return NextResponse.json({ error: 'File mancante' }, { status: 400 })
    }

    const text = await file.text()
    
    const parsed = Papa.parse(text, {
      header: true,
      skipEmptyLines: true
    })

    const imported = []
    const errors = []

    for (let i = 0; i < parsed.data.length; i++) {
      const row: any = parsed.data[i]
      
      try {
        // Validazione base
        if (!row.Nome || !row.Cognome || !row['Codice Fiscale']) {
          errors.push(`Riga ${i + 2}: campi obbligatori mancanti`)
          continue
        }

        // Parse data nascita (formato dd/MM/yyyy)
        const [day, month, year] = row['Data Nascita'].split('/')
        const dataNascita = new Date(parseInt(year), parseInt(month) - 1, parseInt(day))

        const paziente = await prisma.paziente.create({
          data: {
            nome: row.Nome.trim(),
            cognome: row.Cognome.trim(),
            codiceFiscale: row['Codice Fiscale'].trim().toUpperCase(),
            dataNascita,
            luogoNascita: row['Luogo Nascita'] || '',
            sesso: row.Sesso || 'M',
            indirizzo: row.Indirizzo || '',
            citta: row.Città || '',
            provincia: row.Provincia || '',
            cap: row.CAP || '',
            telefono: row.Telefono || '',
            email: row.Email || null,
            note: row.Note || null,
            ambulatorioId: user.ambulatorioId
          }
        })

        imported.push(paziente.id)
      } catch (error: any) {
        if (error.code === 'P2002') {
          errors.push(`Riga ${i + 2}: Codice Fiscale duplicato`)
        } else {
          errors.push(`Riga ${i + 2}: ${error.message}`)
        }
      }
    }

    return NextResponse.json({
      success: true,
      imported: imported.length,
      errors
    })
  } catch (error) {
    console.error('Errore import CSV:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}
```

## 📄 FILE 4: app/dashboard/pazienti/page.tsx

```typescript
'use client'

import { useState, useEffect } from 'react'
import { Search, Filter, Download, Upload, Plus, User, Calendar, MapPin } from 'lucide-react'
import Link from 'next/link'
import { format } from 'date-fns'
import { it } from 'date-fns/locale'

export default function PazientiPage() {
  const [pazienti, setPazienti] = useState<any[]>([])
  const [loading, setLoading] = useState(true)
  const [searchQuery, setSearchQuery] = useState('')
  const [filters, setFilters] = useState({
    sesso: 'all',
    citta: 'all',
    etaMin: 0,
    etaMax: 120,
    ultimaVisita: 'all'
  })
  const [pagination, setPagination] = useState({
    page: 1,
    limit: 20,
    total: 0,
    totalPages: 0
  })
  const [sortBy, setSortBy] = useState('cognome')
  const [sortOrder, setSortOrder] = useState<'asc' | 'desc'>('asc')
  const [showFilters, setShowFilters] = useState(false)
  const [importing, setImporting] = useState(false)

  useEffect(() => {
    fetchPazienti()
  }, [searchQuery, filters, pagination.page, sortBy, sortOrder])

  const fetchPazienti = async () => {
    setLoading(true)
    try {
      const params = new URLSearchParams({
        q: searchQuery,
        sesso: filters.sesso,
        citta: filters.citta,
        etaMin: filters.etaMin.toString(),
        etaMax: filters.etaMax.toString(),
        ultimaVisita: filters.ultimaVisita,
        page: pagination.page.toString(),
        limit: pagination.limit.toString(),
        sortBy,
        sortOrder
      })

      const res = await fetch(`/api/pazienti/search?${params}`)
      const data = await res.json()
      
      setPazienti(data.pazienti)
      setPagination(data.pagination)
    } catch (error) {
      console.error('Errore fetch pazienti:', error)
    } finally {
      setLoading(false)
    }
  }

  const handleExport = async () => {
    window.open('/api/pazienti/export', '_blank')
  }

  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    setImporting(true)
    try {
      const formData = new FormData()
      formData.append('file', file)

      const res = await fetch('/api/pazienti/import', {
        method: 'POST',
        body: formData
      })

      const data = await res.json()
      
      if (data.errors.length > 0) {
        alert(`Importati: ${data.imported}\nErrori: ${data.errors.join('\n')}`)
      } else {
        alert(`✅ Importati ${data.imported} pazienti con successo!`)
      }

      fetchPazienti()
    } catch (error) {
      alert('Errore durante import')
    } finally {
      setImporting(false)
    }
  }

  const handleSort = (column: string) => {
    if (sortBy === column) {
      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc')
    } else {
      setSortBy(column)
      setSortOrder('asc')
    }
  }

  const getStatusBadge = (status: string) => {
    const styles = {
      attivo: 'bg-green-100 text-green-700',
      recente: 'bg-blue-100 text-blue-700',
      inattivo: 'bg-gray-100 text-gray-600'
    }
    const labels = {
      attivo: 'Attivo',
      recente: 'Recente',
      inattivo: 'Inattivo'
    }
    return (
      <span className={`px-2 py-1 rounded-full text-xs font-medium ${styles[status as keyof typeof styles]}`}>
        {labels[status as keyof typeof labels]}
      </span>
    )
  }

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Pazienti</h1>
          <p className="text-gray-600 mt-1">{pagination.total} pazienti totali</p>
        </div>
        <div className="flex gap-3">
          <button
            onClick={handleExport}
            className="flex items-center gap-2 px-4 py-2 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200"
          >
            <Download className="w-4 h-4" />
            Export CSV
          </button>
          <label className="flex items-center gap-2 px-4 py-2 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 cursor-pointer">
            <Upload className="w-4 h-4" />
            {importing ? 'Importazione...' : 'Import CSV'}
            <input
              type="file"
              accept=".csv"
              onChange={handleImport}
              disabled={importing}
              className="hidden"
            />
          </label>
          <Link
            href="/dashboard/pazienti/nuovo"
            className="flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700"
          >
            <Plus className="w-4 h-4" />
            Nuovo Paziente
          </Link>
        </div>
      </div>

      {/* Search & Filters */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-4">
        <div className="flex gap-3 mb-4">
          <div className="flex-1 relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-gray-400" />
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              placeholder="Cerca per nome, cognome, CF, telefono..."
              className="w-full pl-10 pr-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500"
            />
          </div>
          <button
            onClick={() => setShowFilters(!showFilters)}
            className="flex items-center gap-2 px-4 py-2 bg-gray-100 rounded-lg hover:bg-gray-200"
          >
            <Filter className="w-4 h-4" />
            Filtri
          </button>
        </div>

        {showFilters && (
          <div className="grid grid-cols-1 md:grid-cols-4 gap-4 pt-4 border-t">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">Sesso</label>
              <select
                value={filters.sesso}
                onChange={(e) => setFilters({ ...filters, sesso: e.target.value })}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg"
              >
                <option value="all">Tutti</option>
                <option value="M">Maschio</option>
                <option value="F">Femmina</option>
              </select>
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">Età Min</label>
              <input
                type="number"
                value={filters.etaMin}
                onChange={(e) => setFilters({ ...filters, etaMin: parseInt(e.target.value) || 0 })}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">Età Max</label>
              <input
                type="number"
                value={filters.etaMax}
                onChange={(e) => setFilters({ ...filters, etaMax: parseInt(e.target.value) || 120 })}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">Ultima Visita</label>
              <select
                value={filters.ultimaVisita}
                onChange={(e) => setFilters({ ...filters, ultimaVisita: e.target.value })}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg"
              >
                <option value="all">Tutte</option>
                <option value="30">Ultimi 30 giorni</option>
                <option value="90">Ultimi 3 mesi</option>
                <option value="180">Ultimi 6 mesi</option>
                <option value="365">Ultimo anno</option>
              </select>
            </div>
          </div>
        )}
      </div>

      {/* Tabella */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-200 overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full">
            <thead className="bg-gray-50 border-b border-gray-200">
              <tr>
                <th className="px-6 py-3 text-left">
                  <button
                    onClick={() => handleSort('cognome')}
                    className="flex items-center gap-1 text-xs font-medium text-gray-700 uppercase hover:text-gray-900"
                  >
                    Paziente
                    {sortBy === 'cognome' && <span>{sortOrder === 'asc' ? '↑' : '↓'}</span>}
                  </button>
                </th>
                <th className="px-6 py-3 text-left text-xs font-medium text-gray-700 uppercase">
                  Età
                </th>
                <th className="px-6 py-3 text-left text-xs font-medium text-gray-700 uppercase">
                  Contatti
                </th>
                <th className="px-6 py-3 text-left text-xs font-medium text-gray-700 uppercase">
                  Località
                </th>
                <th className="px-6 py-3 text-left">
                  <button
                    onClick={() => handleSort('ultimaCartella')}
                    className="flex items-center gap-1 text-xs font-medium text-gray-700 uppercase hover:text-gray-900"
                  >
                    Ultima Visita
                    {sortBy === 'ultimaCartella' && <span>{sortOrder === 'asc' ? '↑' : '↓'}</span>}
                  </button>
                </th>
                <th className="px-6 py-3 text-left text-xs font-medium text-gray-700 uppercase">
                  Status
                </th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-200">
              {loading ? (
                <tr>
                  <td colSpan={6} className="px-6 py-12 text-center text-gray-500">
                    Caricamento...
                  </td>
                </tr>
              ) : pazienti.length === 0 ? (
                <tr>
                  <td colSpan={6} className="px-6 py-12 text-center text-gray-500">
                    Nessun paziente trovato
                  </td>
                </tr>
              ) : (
                pazienti.map((p) => (
                  <tr key={p.id} className="hover:bg-gray-50">
                    <td className="px-6 py-4">
                      <Link
                        href={`/dashboard/pazienti/${p.id}`}
                        className="flex items-center gap-3 group"
                      >
                        <div className="w-10 h-10 bg-blue-100 rounded-full flex items-center justify-center">
                          <User className="w-5 h-5 text-blue-600" />
                        </div>
                        <div>
                          <div className="flex items-center gap-2">
                            <span className="font-medium text-gray-900 group-hover:text-blue-600">
                              {p.cognome} {p.nome}
                            </span>
                            {p.isNuovo && (
                              <span className="px-2 py-0.5 bg-green-100 text-green-700 text-xs font-semibold rounded">
                                Nuovo
                              </span>
                            )}
                          </div>
                          <span className="text-sm text-gray-500">{p.codiceFiscale}</span>
                        </div>
                      </Link>
                    </td>
                    <td className="px-6 py-4 text-sm text-gray-900">
                      {p.eta} anni
                    </td>
                    <td className="px-6 py-4">
                      <div className="text-sm">
                        <div className="text-gray-900">{p.telefono}</div>
                        {p.email && <div className="text-gray-500">{p.email}</div>}
                      </div>
                    </td>
                    <td className="px-6 py-4">
                      <div className="flex items-center gap-1 text-sm text-gray-900">
                        <MapPin className="w-4 h-4 text-gray-400" />
                        {p.citta} ({p.provincia})
                      </div>
                    </td>
                    <td className="px-6 py-4">
                      {p.ultimaCartella ? (
                        <div className="flex items-center gap-1 text-sm text-gray-900">
                          <Calendar className="w-4 h-4 text-gray-400" />
                          {format(new Date(p.ultimaCartella), 'dd/MM/yyyy', { locale: it })}
                        </div>
                      ) : (
                        <span className="text-sm text-gray-500">Mai visitato</span>
                      )}
                    </td>
                    <td className="px-6 py-4">
                      {getStatusBadge(p.status)}
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>

        {/* Paginazione */}
        {pagination.totalPages > 1 && (
          <div className="px-6 py-4 border-t border-gray-200 flex items-center justify-between">
            <div className="text-sm text-gray-700">
              Pagina {pagination.page} di {pagination.totalPages} ({pagination.total} totali)
            </div>
            <div className="flex gap-2">
              <button
                onClick={() => setPagination({ ...pagination, page: pagination.page - 1 })}
                disabled={pagination.page === 1}
                className="px-4 py-2 border border-gray-300 rounded-lg hover:bg-gray-50 disabled:opacity-50"
              >
                Precedente
              </button>
              <button
                onClick={() => setPagination({ ...pagination, page: pagination.page + 1 })}
                disabled={pagination.page === pagination.totalPages}
                className="px-4 py-2 border border-gray-300 rounded-lg hover:bg-gray-50 disabled:opacity-50"
              >
                Successiva
              </button>
            </div>
          </div>
        )}
      </div>
    </div>
  )
}
```
