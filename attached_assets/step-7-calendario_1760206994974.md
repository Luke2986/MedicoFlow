import { format, startOfWeek, endOfWeek, addWeeks, subWeeks, addDays, isSameDay } from 'date-fns'
import { it } from 'date-fns/locale'

export function getWeekDays(date: Date) {
  const start = startOfWeek(date, { weekStartsOn: 1 }) // lunedì
  return Array.from({ length: 7 }, (_, i) => addDays(start, i))
}

export function getWeekRange(date: Date) {
  const start = startOfWeek(date, { weekStartsOn: 1 })
  const end = endOfWeek(date, { weekStartsOn: 1 })
  return { start, end }
}

export function formatWeekRange(date: Date) {
  const { start, end } = getWeekRange(date)
  return `${format(start, 'd MMM', { locale: it })} - ${format(end, 'd MMM yyyy', { locale: it })}`
}

export function getTimeSlots(startHour = 8, endHour = 20, interval = 30) {
  const slots = []
  for (let hour = startHour; hour < endHour; hour++) {
    for (let minutes = 0; minutes < 60; minutes += interval) {
      slots.push({
        hour,
        minutes,
        label: `${hour.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}`
      })
    }
  }
  return slots
}

export function getAppuntamentoStyle(stato: string) {
  const styles = {
    confermato: 'bg-blue-500 border-blue-600 text-white',
    completato: 'bg-green-500 border-green-600 text-white',
    cancellato: 'bg-gray-400 border-gray-500 text-white',
    'no-show': 'bg-red-500 border-red-600 text-white'
  }
  return styles[stato as keyof typeof styles] || styles.confermato
}

export function checkConflict(
  newStart: Date,
  newEnd: Date,
  existingAppuntamenti: any[],
  excludeId?: string
) {
  return existingAppuntamenti.some(apt => {
    if (apt.id === excludeId) return false
    
    const aptStart = new Date(apt.dataOra)
    const aptEnd = new Date(aptStart.getTime() + apt.durata * 60000)
    
    return (newStart < aptEnd && newEnd > aptStart)
  })
}

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { startOfWeek, endOfWeek } from 'date-fns'

export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const searchParams = request.nextUrl.searchParams
    const dateParam = searchParams.get('date')
    const date = dateParam ? new Date(dateParam) : new Date()
    
    const start = startOfWeek(date, { weekStartsOn: 1 })
    const end = endOfWeek(date, { weekStartsOn: 1 })

    const appuntamenti = await prisma.appuntamento.findMany({
      where: {
        ambulatorioId: user.ambulatorioId,
        dataOra: {
          gte: start,
          lte: end
        }
      },
      include: {
        paziente: {
          select: {
            id: true,
            nome: true,
            cognome: true,
            telefono: true
          }
        }
      },
      orderBy: { dataOra: 'asc' }
    })

    return NextResponse.json(appuntamenti)
  } catch (error) {
    console.error('Errore GET appuntamenti:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const body = await request.json()
    
    // Check conflicts
    const existingAppuntamenti = await prisma.appuntamento.findMany({
      where: {
        ambulatorioId: user.ambulatorioId,
        medicoId: user.id,
        dataOra: {
          gte: new Date(new Date(body.dataOra).getTime() - 2 * 60 * 60 * 1000),
          lte: new Date(new Date(body.dataOra).getTime() + 2 * 60 * 60 * 1000)
        }
      }
    })

    const newEnd = new Date(new Date(body.dataOra).getTime() + body.durata * 60000)
    const hasConflict = existingAppuntamenti.some(apt => {
      const aptStart = new Date(apt.dataOra)
      const aptEnd = new Date(aptStart.getTime() + apt.durata * 60000)
      return (new Date(body.dataOra) < aptEnd && newEnd > aptStart)
    })

    if (hasConflict) {
      return NextResponse.json(
        { error: 'Conflitto: esiste già un appuntamento in questo orario' },
        { status: 409 }
      )
    }

    const appuntamento = await prisma.appuntamento.create({
      data: {
        pazienteId: body.pazienteId,
        medicoId: user.id,
        dataOra: new Date(body.dataOra),
        durata: body.durata || 30,
        tipo: body.tipo,
        stato: 'confermato',
        motivoVisita: body.motivoVisita,
        note: body.note,
        ambulatorioId: user.ambulatorioId
      },
      include: {
        paziente: {
          select: {
            id: true,
            nome: true,
            cognome: true,
            telefono: true
          }
        }
      }
    })

    return NextResponse.json(appuntamento, { status: 201 })
  } catch (error) {
    console.error('Errore POST appuntamento:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'

export async function PUT(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    const body = await request.json()

    const appuntamento = await prisma.appuntamento.update({
      where: { id: params.id },
      data: {
        dataOra: body.dataOra ? new Date(body.dataOra) : undefined,
        durata: body.durata,
        tipo: body.tipo,
        stato: body.stato,
        motivoVisita: body.motivoVisita,
        note: body.note
      },
      include: {
        paziente: {
          select: {
            id: true,
            nome: true,
            cognome: true,
            telefono: true
          }
        }
      }
    })

    return NextResponse.json(appuntamento)
  } catch (error) {
    console.error('Errore PUT appuntamento:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

export async function DELETE(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await getCurrentUser()
    if (!user) {
      return NextResponse.json({ error: 'Non autorizzato' }, { status: 401 })
    }

    await prisma.appuntamento.delete({
      where: { id: params.id }
    })

    return NextResponse.json({ success: true })
  } catch (error) {
    console.error('Errore DELETE appuntamento:', error)
    return NextResponse.json({ error: 'Errore server' }, { status: 500 })
  }
}

'use client'

import { useState } from 'react'
import { ChevronLeft, ChevronRight } from 'lucide-react'
import { format, addWeeks, subWeeks, isSameDay, isToday } from 'date-fns'
import { it } from 'date-fns/locale'
import { getWeekDays, formatWeekRange, getTimeSlots, getAppuntamentoStyle } from '@/lib/calendar-utils'

interface WeekCalendarProps {
  appuntamenti: any[]
  onAppuntamentoClick: (appuntamento: any) => void
  onSlotClick: (date: Date, hour: number, minutes: number) => void
  onWeekChange: (date: Date) => void
}

export default function WeekCalendar({
  appuntamenti,
  onAppuntamentoClick,
  onSlotClick,
  onWeekChange
}: WeekCalendarProps) {
  const [currentDate, setCurrentDate] = useState(new Date())
  const weekDays = getWeekDays(currentDate)
  const timeSlots = getTimeSlots(8, 20, 30)

  const handlePrevWeek = () => {
    const newDate = subWeeks(currentDate, 1)
    setCurrentDate(newDate)
    onWeekChange(newDate)
  }

  const handleNextWeek = () => {
    const newDate = addWeeks(currentDate, 1)
    setCurrentDate(newDate)
    onWeekChange(newDate)
  }

  const handleToday = () => {
    const today = new Date()
    setCurrentDate(today)
    onWeekChange(today)
  }

  const getAppuntamentiForDayAndSlot = (day: Date, hour: number, minutes: number) => {
    return appuntamenti.filter(apt => {
      const aptDate = new Date(apt.dataOra)
      return (
        isSameDay(aptDate, day) &&
        aptDate.getHours() === hour &&
        aptDate.getMinutes() === minutes
      )
    })
  }

  return (
    <div className="bg-white rounded-lg shadow-sm border border-gray-200 overflow-hidden">
      {/* Header */}
      <div className="flex items-center justify-between p-4 border-b border-gray-200 bg-gray-50">
        <div className="flex items-center gap-4">
          <button
            onClick={handlePrevWeek}
            className="p-2 hover:bg-gray-200 rounded-lg transition-colors"
          >
            <ChevronLeft className="w-5 h-5" />
          </button>
          <button
            onClick={handleNextWeek}
            className="p-2 hover:bg-gray-200 rounded-lg transition-colors"
          >
            <ChevronRight className="w-5 h-5" />
          </button>
          <h2 className="text-xl font-semibold text-gray-900">
            {formatWeekRange(currentDate)}
          </h2>
        </div>
        <button
          onClick={handleToday}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors"
        >
          Oggi
        </button>
      </div>

      {/* Calendar Grid */}
      <div className="overflow-x-auto">
        <div className="min-w-[800px]">
          {/* Days Header */}
          <div className="grid grid-cols-8 border-b border-gray-200 bg-gray-50 sticky top-0 z-10">
            <div className="p-2 text-sm font-medium text-gray-500 border-r border-gray-200">
              Orario
            </div>
            {weekDays.map((day, idx) => (
              <div
                key={idx}
                className={`p-2 text-center border-r border-gray-200 last:border-r-0 ${
                  isToday(day) ? 'bg-blue-50' : ''
                }`}
              >
                <div className="text-sm font-medium text-gray-500">
                  {format(day, 'EEE', { locale: it })}
                </div>
                <div className={`text-lg font-semibold ${
                  isToday(day) ? 'text-blue-600' : 'text-gray-900'
                }`}>
                  {format(day, 'd')}
                </div>
              </div>
            ))}
          </div>

          {/* Time Slots */}
          <div className="relative">
            {timeSlots.map((slot, slotIdx) => (
              <div
                key={slotIdx}
                className="grid grid-cols-8 border-b border-gray-200 last:border-b-0"
                style={{ minHeight: '60px' }}
              >
                {/* Time Label */}
                <div className="p-2 text-sm font-medium text-gray-500 border-r border-gray-200 flex items-start">
                  {slot.label}
                </div>

                {/* Day Cells */}
                {weekDays.map((day, dayIdx) => {
                  const dayAppuntamenti = getAppuntamentiForDayAndSlot(day, slot.hour, slot.minutes)
                  
                  return (
                    <div
                      key={dayIdx}
                      onClick={() => dayAppuntamenti.length === 0 && onSlotClick(day, slot.hour, slot.minutes)}
                      className={`relative border-r border-gray-200 last:border-r-0 hover:bg-blue-50 cursor-pointer transition-colors ${
                        isToday(day) ? 'bg-blue-50/30' : ''
                      }`}
                    >
                      {dayAppuntamenti.map((apt) => (
                        <div
                          key={apt.id}
                          onClick={(e) => {
                            e.stopPropagation()
                            onAppuntamentoClick(apt)
                          }}
                          className={`absolute inset-x-1 rounded-md border-l-4 p-2 text-xs cursor-pointer hover:shadow-md transition-shadow ${getAppuntamentoStyle(apt.stato)}`}
                          style={{
                            top: '4px',
                            height: `${(apt.durata / 30) * 60 - 8}px`,
                            zIndex: 5
                          }}
                        >
                          <div className="font-semibold truncate">
                            {apt.paziente.nome} {apt.paziente.cognome}
                          </div>
                          <div className="truncate opacity-90">
                            {apt.motivoVisita || apt.tipo}
                          </div>
                          <div className="text-xs opacity-75 mt-1">
                            {apt.durata} min
                          </div>
                        </div>
                      ))}
                    </div>
                  )
                })}
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* Legend */}
      <div className="flex items-center gap-4 p-4 border-t border-gray-200 bg-gray-50 text-sm">
        <span className="font-medium text-gray-700">Legenda:</span>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded bg-blue-500"></div>
          <span className="text-gray-600">Confermato</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded bg-green-500"></div>
          <span className="text-gray-600">Completato</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded bg-red-500"></div>
          <span className="text-gray-600">No-show</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded bg-gray-400"></div>
          <span className="text-gray-600">Cancellato</span>
        </div>
      </div>
    </div>
  )
}

'use client'

import { useState, useEffect } from 'react'
import { X, Loader2, Trash2, Calendar, Clock, User } from 'lucide-react'
import { format } from 'date-fns'

interface AppointmentModalProps {
  isOpen: boolean
  onClose: () => void
  appuntamento?: any
  defaultDate?: Date
  defaultTime?: { hour: number; minutes: number }
  pazienti: any[]
  onSave: (data: any) => Promise<void>
  onDelete?: (id: string) => Promise<void>
}

export default function AppointmentModal({
  isOpen,
  onClose,
  appuntamento,
  defaultDate,
  defaultTime,
  pazienti,
  onSave,
  onDelete
}: AppointmentModalProps) {
  const [loading, setLoading] = useState(false)
  const [formData, setFormData] = useState({
    pazienteId: '',
    data: '',
    ora: '',
    durata: 30,
    tipo: 'controllo',
    motivoVisita: '',
    note: '',
    stato: 'confermato'
  })

  useEffect(() => {
    if (appuntamento) {
      const aptDate = new Date(appuntamento.dataOra)
      setFormData({
        pazienteId: appuntamento.pazienteId,
        data: format(aptDate, 'yyyy-MM-dd'),
        ora: format(aptDate, 'HH:mm'),
        durata: appuntamento.durata,
        tipo: appuntamento.tipo,
        motivoVisita: appuntamento.motivoVisita || '',
        note: appuntamento.note || '',
        stato: appuntamento.stato
      })
    } else if (defaultDate && defaultTime) {
      setFormData(prev => ({
        ...prev,
        data: format(defaultDate, 'yyyy-MM-dd'),
        ora: `${defaultTime.hour.toString().padStart(2, '0')}:${defaultTime.minutes.toString().padStart(2, '0')}`
      }))
    }
  }, [appuntamento, defaultDate, defaultTime])

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setLoading(true)

    try {
      const [hours, minutes] = formData.ora.split(':').map(Number)
      const dataOra = new Date(formData.data)
      dataOra.setHours(hours, minutes, 0, 0)

      await onSave({
        id: appuntamento?.id,
        pazienteId: formData.pazienteId,
        dataOra: dataOra.toISOString(),
        durata: formData.durata,
        tipo: formData.tipo,
        motivoVisita: formData.motivoVisita,
        note: formData.note,
        stato: formData.stato
      })

      onClose()
    } catch (error) {
      console.error('Errore salvataggio:', error)
      alert('Errore durante il salvataggio')
    } finally {
      setLoading(false)
    }
  }

  const handleDelete = async () => {
    if (!appuntamento || !onDelete) return
    if (!confirm('Sei sicuro di voler eliminare questo appuntamento?')) return

    setLoading(true)
    try {
      await onDelete(appuntamento.id)
      onClose()
    } catch (error) {
      alert('Errore durante l\'eliminazione')
    } finally {
      setLoading(false)
    }
  }

  if (!isOpen) return null

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-xl shadow-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto">
        {/* Header */}
        <div className="flex items-center justify-between p-6 border-b border-gray-200">
          <h2 className="text-2xl font-bold text-gray-900">
            {appuntamento ? 'Modifica Appuntamento' : 'Nuovo Appuntamento'}
          </h2>
          <button
            onClick={onClose}
            className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Form */}
        <form onSubmit={handleSubmit} className="p-6 space-y-6">
          {/* Paziente */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              <User className="w-4 h-4 inline mr-2" />
              Paziente *
            </label>
            <select
              value={formData.pazienteId}
              onChange={(e) => setFormData(prev => ({ ...prev, pazienteId: e.target.value }))}
              required
              className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            >
              <option value="">Seleziona paziente...</option>
              {pazienti.map(p => (
                <option key={p.id} value={p.id}>
                  {p.cognome} {p.nome}
                </option>
              ))}
            </select>
          </div>

          {/* Data e Ora */}
          <div className="grid grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                <Calendar className="w-4 h-4 inline mr-2" />
                Data *
              </label>
              <input
                type="date"
                value={formData.data}
                onChange={(e) => setFormData(prev => ({ ...prev, data: e.target.value }))}
                required
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                <Clock className="w-4 h-4 inline mr-2" />
                Ora *
              </label>
              <input
                type="time"
                value={formData.ora}
                onChange={(e) => setFormData(prev => ({ ...prev, ora: e.target.value }))}
                required
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>
          </div>

          {/* Durata e Tipo */}
          <div className="grid grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                Durata (minuti) *
              </label>
              <select
                value={formData.durata}
                onChange={(e) => setFormData(prev => ({ ...prev, durata: Number(e.target.value) }))}
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              >
                <option value={15}>15 minuti</option>
                <option value={30}>30 minuti</option>
                <option value={45}>45 minuti</option>
                <option value={60}>60 minuti</option>
                <option value={90}>90 minuti</option>
              </select>
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                Tipo visita *
              </label>
              <select
                value={formData.tipo}
                onChange={(e) => setFormData(prev => ({ ...prev, tipo: e.target.value }))}
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              >
                <option value="prima-visita">Prima visita</option>
                <option value="controllo">Controllo</option>
                <option value="urgente">Urgente</option>
                <option value="telemedicina">Telemedicina</option>
              </select>
            </div>
          </div>

          {/* Stato (solo se modifica) */}
          {appuntamento && (
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                Stato
              </label>
              <select
                value={formData.stato}
                onChange={(e) => setFormData(prev => ({ ...prev, stato: e.target.value }))}
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              >
                <option value="confermato">Confermato</option>
                <option value="completato">Completato</option>
                <option value="cancellato">Cancellato</option>
                <option value="no-show">No-show</option>
              </select>
            </div>
          )}

          {/* Motivo */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Motivo visita
            </label>
            <input
              type="text"
              value={formData.motivoVisita}
              onChange={(e) => setFormData(prev => ({ ...prev, motivoVisita: e.target.value }))}
              placeholder="Es: Controllo periodico"
              className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          {/* Note */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Note
            </label>
            <textarea
              value={formData.note}
              onChange={(e) => setFormData(prev => ({ ...prev, note: e.target.value }))}
              rows={3}
              placeholder="Note aggiuntive..."
              className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          {/* Actions */}
          <div className="flex gap-3">
            <button
              type="submit"
              disabled={loading}
              className="flex-1 flex items-center justify-center gap-2 px-6 py-3 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50 transition-colors"

              {appuntamento ? 'Salva Modifiche' : 'Crea Appuntamento'}
            </button>

              <button
                type="button"
                onClick={handleDelete}
                disabled={loading}
                className="px-6 py-3 bg-red-600 text-white rounded-lg hover:bg-red-700 disabled:opacity-50 transition-colors flex items-center gap-2"
              >
                <Trash2 className="w-4 h-4" />
                Elimina
              </button>
            )}
            
            <button
              type="button"
              onClick={onClose}
              className="px-6 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
            >
              Annulla
            </button>
          </div>
        </form>
      </div>
    </div>

## 📁 FILE 6: app/dashboard/appuntamenti/page.tsx

```typescript
'use client'

import { useState, useEffect } from 'react'
import { Plus } from 'lucide-react'
import WeekCalendar from '@/components/calendario/WeekCalendar'
import AppointmentModal from '@/components/calendario/AppointmentModal'

export default function AppuntamentiPage() {
  const [appuntamenti, setAppuntamenti] = useState([])
  const [pazienti, setPazienti] = useState([])
  const [loading, setLoading] = useState(true)
  const [modalOpen, setModalOpen] = useState(false)
  const [selectedAppuntamento, setSelectedAppuntamento] = useState<any>(null)
  const [selectedSlot, setSelectedSlot] = useState<any>(null)

  const fetchAppuntamenti = async (date: Date = new Date()) => {
    try {
      const res = await fetch(`/api/appuntamenti?date=${date.toISOString()}`)
      const data = await res.json()
      setAppuntamenti(data)
    } catch (error) {
      console.error('Errore fetch appuntamenti:', error)
    }
  }

  const fetchPazienti = async () => {
    try {
      const res = await fetch('/api/pazienti')
      const data = await res.json()
      setPazienti(data)
    } catch (error) {
      console.error('Errore fetch pazienti:', error)
    }
  }

  useEffect(() => {
    Promise.all([fetchAppuntamenti(), fetchPazienti()]).finally(() => setLoading(false))
  }, [])

  const handleAppuntamentoClick = (appuntamento: any) => {
    setSelectedAppuntamento(appuntamento)
    setSelectedSlot(null)
    setModalOpen(true)
  }

  const handleSlotClick = (date: Date, hour: number, minutes: number) => {
    setSelectedSlot({ date, hour, minutes })
    setSelectedAppuntamento(null)
    setModalOpen(true)
  }

  const handleSave = async (data: any) => {
    try {
      const url = data.id ? `/api/appuntamenti/${data.id}` : '/api/appuntamenti'
      const method = data.id ? 'PUT' : 'POST'

      const res = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      })

      if (res.status === 409) {
        alert('Conflitto: esiste già un appuntamento in questo orario')
        return
      }

      if (!res.ok) throw new Error('Errore salvataggio')

      await fetchAppuntamenti()
      setModalOpen(false)
    } catch (error) {
      throw error
    }
  }

  const handleDelete = async (id: string) => {
    try {
      const res = await fetch(`/api/appuntamenti/${id}`, { method: 'DELETE' })
      if (!res.ok) throw new Error('Errore eliminazione')
      
      await fetchAppuntamenti()
    } catch (error) {
      throw error
    }
  }

  const handleWeekChange = (date: Date) => {
    fetchAppuntamenti(date)
  }

  if (loading) {
    return (
      <div className="flex items-center justify-center h-96">
        <div className="text-gray-500">Caricamento calendario...</div>
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Calendario Appuntamenti</h1>
          <p className="text-gray-600 mt-1">Gestisci gli appuntamenti settimanali</p>
        </div>
        <button
          onClick={() => {
            setSelectedAppuntamento(null)
            setSelectedSlot({ date: new Date(), hour: 9, minutes: 0 })
            setModalOpen(true)
          }}
          className="flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors"
        >
          <Plus className="w-4 h-4" />
          Nuovo Appuntamento
        </button>
      </div>

      <WeekCalendar
        appuntamenti={appuntamenti}
        onAppuntamentoClick={handleAppuntamentoClick}
        onSlotClick={handleSlotClick}
        onWeekChange={handleWeekChange}
      />

      <AppointmentModal
        isOpen={modalOpen}
        onClose={() => setModalOpen(false)}
        appuntamento={selectedAppuntamento}
        defaultDate={selectedSlot?.date}
        defaultTime={selectedSlot ? { hour: selectedSlot.hour, minutes: selectedSlot.minutes } : undefined}
        pazienti={pazienti}
        onSave={handleSave}
        onDelete={handleDelete}
      />

      <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
        <h3 className="font-semibold text-gray-900 mb-2">💡 Suggerimenti</h3>
        <ul className="text-sm text-gray-700 space-y-1">
          <li>• Clicca su uno slot vuoto per creare un nuovo appuntamento</li>
          <li>• Clicca su un appuntamento esistente per modificarlo o eliminarlo</li>
          <li>• Usa i pulsanti freccia per navigare tra le settimane</li>
          <li>• Il sistema previene automaticamente i conflitti di orario</li>
        </ul>
      </div>
    </div>
  )
}
```
